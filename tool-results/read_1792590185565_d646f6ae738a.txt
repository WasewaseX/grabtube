   106→export function runYtDlp(args: string[], timeoutMs = 120000): Promise<RunResult> {
   107→  return new Promise((resolve, reject) => {
   108→    execFile(
   109→      getYtDlpPath(),
   110→      args,
   111→      { timeout: timeoutMs, maxBuffer: 64 * 1024 * 1024, env: toolEnv() },
   112→      (err, stdout, stderr) => {
   113→        if (err && typeof (err as NodeJS.ErrnoException).code === 'undefined' && !stdout && !stderr) {
   114→          return reject(err)
   115→        }
   116→        resolve({ code: err ? 1 : 0, stdout: stdout ?? '', stderr: stderr ?? '' })
   117→      }
   118→    )
   119→  })
   120→}

   211→}

   287→// ---- Info store ------------------------------------------------------------
   288→
   289→interface StoreEntry {
   290→  raw: RawInfo
   291→  at: number
   292→}
   293→
   294→declare global {
   295→  var __tgInfoStore: Map<string, StoreEntry> | undefined
   296→}
   297→
   298→function infoStore(): Map<string, StoreEntry> {
   299→  if (!globalThis.__tgInfoStore) globalThis.__tgInfoStore = new Map()
   300→  return globalThis.__tgInfoStore
   301→}
   302→
   303→const INFO_TTL_MS = 6 * 60 * 60 * 1000
   304→const INFO_MAX = 60
   305→
   306→function pruneStore(): void {
   307→  const store = infoStore()
   308→  const now = Date.now()
   309→  for (const [k, v] of store) {
   310→    if (now - v.at > INFO_TTL_MS) store.delete(k)
   311→  }
   312→  while (store.size > INFO_MAX) {
   313→    const oldest = [...store.entries()].sort((a, b) => a[1].at - b[1].at)[0]
   314→    if (!oldest) break
   315→    store.delete(oldest[0])
   316→  }
   317→}
   318→
   319→export function putInfo(raw: RawInfo): string {
   320→  const id = randomUUID()
   321→  const store = infoStore()
   322→  store.set(id, { raw, at: Date.now() })
   323→  pruneStore()
   324→  return id
   325→}
   326→
   327→export function getInfo(infoId: string): StoreEntry | null {
   328→  const e = infoStore().get(infoId)
   329→  if (!e) return null
   330→  if (Date.now() - e.at > INFO_TTL_MS) {
   331→    infoStore().delete(infoId)
   332→    return null
   333→  }
   334→  return e
   335→}

   397→    const { proxiedFetch } = await import('./proxy')
   398→    const res = await proxiedFetch('https://www.youtube.com/', {

   442→  }

   492→async function parseRaw(stdout: string, url: string): Promise<RawInfo> {
   493→  const data = JSON.parse(stdout) as Record<string, unknown>
   494→  const formats = ((data.formats as YtRawFormat[]) ?? []).filter((f) => {
   495→    if (!f.format_id) return false
   496→    if ((f.ext ?? '') === 'mhtml') return false
   497→    if (/storyboard/i.test(f.format_note ?? '')) return false
   498→    return true
   499→  })
   500→  return {
   501→    id: String(data.id ?? extractVideoId(url) ?? ''),
   502→    title: String(data.title ?? 'Untitled'),
   503→    uploader: (data.uploader as string) || (data.channel as string) || undefined,
   504→    channel: (data.channel as string) || undefined,
   505→    duration: (data.duration as number) || undefined,
   506→    view_count: (data.view_count as number) || undefined,
   507→    thumbnail: (data.thumbnail as string) || undefined,
   508→    webpage_url: String(data.webpage_url ?? url),
   509→    is_live: Boolean(data.is_live),
   510→    live_status: (data.live_status as string) || undefined,
   511→    extractor_key: String(data.extractor_key ?? data.extractor ?? 'Generic'),
   512→    formats,
   513→  }
   514→}

   516→/**
   517→ * Full extraction through the current engine stack.
   518→ * Throws DlError with a classified code; extract.ts decides the fallback.
   519→ */
   520→export async function fetchInfo(url: string): Promise<{ infoId: string; raw: RawInfo }> {
   521→  const clean = url.trim()
   522→  if (!/^https?:\/\//i.test(clean)) {
   523→    throw new DlError('INVALID_URL', 'That does not look like a valid video URL.')
   524→  }
   525→
   526→  const { ensurePotServer } = await import('./pot')
   527→  void ensurePotServer().catch(() => undefined)
   528→
   529→  const cookies = await loadCookiesText()
   530→
   531→  if (!isYouTubeUrl(clean)) {
   532→    const { code, stdout, stderr } = await runYtDlp(
   533→      ['-J', ...baseArgs(), ...(cookies ? ['--cookies', cookies] : []), clean],
   534→      150000,
   535→    )
   536→    if (code !== 0 || !stdout.trim().startsWith('{')) {
   537→      const c = classifyStderr(stderr)
   538→      throw new DlError(c.code, c.message, c.hint)
   539→    }
   540→    const raw = await parseRaw(stdout, clean)
   541→    if (raw.is_live) throw new DlError('LIVE', 'Live streams cannot be downloaded.')
   542→    if (raw.formats.length === 0) throw new DlError('NO_FORMATS', 'No downloadable formats found.')
   543→    const infoId = putInfo(raw)
   544→    return { infoId, raw }
   545→  }
   546→
   547→  let visitor = await ensureVisitorData()
   548→  refreshClientChain(visitor)
   549→  let rotatedVisitor = false
   550→  const errors: Classified[] = []
   551→  const MAX_STEPS = 4
   552→
   553→  for (let i = 0; i < Math.min(MAX_STEPS, CLIENT_CHAIN.length); i++) {
   554→    const step = CLIENT_CHAIN[i]
   555→    const { code, stdout, stderr } = await runYtDlp(
   556→      ['-J', ...baseArgs(), ...step, ...(cookies ? ['--cookies', cookies] : []), clean],
   557→      90000,
   558→    )
   559→    if (code === 0 && stdout.trim().startsWith('{')) {
   560→      const raw = await parseRaw(stdout, clean)
   561→      if (raw.is_live) throw new DlError('LIVE', 'Live streams cannot be downloaded.')
   562→      if (raw.formats.length > 0) {
   563→        const infoId = putInfo(raw)
   564→        return { infoId, raw }
   565→      }
   566→      errors.push({ code: 'NO_FORMATS', message: 'Player returned no media formats.' })
   567→      continue
   568→    }
   569→    const c = classifyStderr(stderr || 'empty output')
   570→    if (TERMINAL_CODES.has(c.code)) throw new DlError(c.code, c.message, c.hint)
   571→    errors.push(c)
   572→    // bot wall → rotate visitor identity exactly once and keep rotating clients
   573→    if ((c.code === 'BOT_CHECK' || c.code === 'RATE_LIMITED') && !rotatedVisitor) {
   574→      rotatedVisitor = true
   575→      const minted = await mintVisitorData()
   576→      if (minted && minted !== visitor) {
   577→        visitor = minted
   578→        persistVisitorData(minted)
   579→        refreshClientChain(visitor)
   580→      }
   581→    }
   582→  }
   583→
   584→  const worst = errors.find((e) => e.code === 'BOT_CHECK') ?? errors[0] ?? { code: 'FETCH_FAILED' as const, message: 'Extraction failed.' }
   585→  throw new DlError(worst.code, worst.message, worst.hint)
   586→}

   599→export async function resolveFormatUrl(webpageUrl: string, selector: string): Promise<string | null> {
   600→  try {
   601→    const cookies = await loadCookiesText()
   602→    const { code, stdout } = await runYtDlp(
   603→      [
   604→        ...baseArgs(),
   605→        ...(cookies ? ['--cookies', cookies] : []),
   606→        ...CLIENT_CHAIN[0],
   607→        '-f',
   608→        selector,
   609→        '--get-url',
   610→        webpageUrl,
   611→      ],
   612→      90000,
   613→    )
   614→    if (code !== 0) return null
   615→    const line = stdout
   616→      .split('\n')
   617→      .map((l) => l.trim())
   618→      .find((l) => /^https?:\/\//.test(l))
   619→    return line ?? null
   620→  } catch {
   621→    return null
   622→  }
   623→}

   627→function codecLabel(c?: string): string | undefined {
   628→  if (!c || c === 'none') return undefined
   629→  const s = c.toLowerCase()
   630→  if (s.startsWith('avc1') || s.startsWith('avc3') || s.includes('h264') || s.includes('h.264')) return 'H.264'
   631→  if (s.startsWith('vp9') || s.includes('vp09')) return 'VP9'
   632→  if (s.startsWith('vp8')) return 'VP8'
   633→  if (s.startsWith('av01')) return 'AV1'
   634→  if (s.startsWith('mp4a') || s.includes('aac')) return 'AAC'
   635→  if (s.includes('opus')) return 'Opus'
   636→  if (s.includes('vorbis')) return 'Vorbis'
   637→  if (s.includes('ec-3') || s.includes('eac3')) return 'E-AC3'
   638→  if (s.includes('ac-3')) return 'AC3'
   639→  return c.split('.')[0].toUpperCase()
   640→}

   642→function fmtSize(f: YtRawFormat): number | undefined {
   643→  return f.filesize || f.filesize_approx || undefined
   644→}

   646→function videoQuality(f: YtRawFormat): string {
   647→  const h = f.height
   648→  if (!h) return f.format_note || f.format_id
   649→  const fps = f.fps && f.fps > 30 ? f.fps : 0
   650→  return `${h}p${fps || ''}`
   651→}
   652→
   653→function audioQuality(f: YtRawFormat): string {
   654→  const k = Math.round(f.abr || f.tbr || 0)
   655→  return k > 0 ? `${k}k` : f.format_note || f.format_id
   656→}

   658→function isStreamable(f: YtRawFormat): boolean {
   659→  const proto = (f.protocol ?? '').toLowerCase()
   660→  return proto === 'https' || proto === 'http' || proto === ''
   661→}
   662→
   663→function pickAudio(formats: YtRawFormat[], container: string): YtRawFormat | null {
   664→  const audios = formats.filter((f) => f.vcodec === 'none' && f.acodec && f.acodec !== 'none')
   665→  if (audios.length === 0) return null
   666→  const family = /webm|mkv/.test(container) ? ['opus', 'vorbis'] : ['mp4a', 'aac']
   667→  const scored = audios
   668→    .map((f) => {
   669→      const s = (f.acodec ?? '').toLowerCase()
   670→      const fam = family.some((x) => s.includes(x)) ? 1 : 0
   671→      return { f, score: fam * 1000 + (f.abr ?? f.tbr ?? 0) }
   672→    })
   673→    .sort((a, b) => b.score - a.score)
   674→  return scored[0].f
   675→}

   677→export function toPublicInfo(infoId: string, raw: RawInfo): PublicInfo {
   678→  const isYT = /youtube/i.test(raw.extractor_key ?? '') || isYouTubeUrl(raw.webpage_url)
   679→  const source = /:piped/i.test(raw.extractor_key ?? '')
   680→    ? 'Piped'
   681→    : /:invidious/i.test(raw.extractor_key ?? '')
   682→      ? 'Invidious'
   683→      : isYT
   684→        ? 'youtube'
   685→        : 'other'
   686→
   687→  const thumbProxy = (u?: string): string => {
   688→    if (!u || !/^https?:\/\//.test(u)) return ''
   689→    return `/api/thumb?url=${encodeURIComponent(u)}`
   690→  }
   691→
   692→  const instant: PublicFormat[] = []
   693→  const videoOnly: Array<{ f: YtRawFormat; fmt: PublicFormat }> = []
   694→  const audio: PublicFormat[] = []
   695→
   696→  for (const f of raw.formats) {
   697→    const hasV = f.vcodec && f.vcodec !== 'none'
   698→    const hasA = f.acodec && f.acodec !== 'none'
   699→    const ext = (f.ext ?? 'mp4').toLowerCase()
   700→    const size = fmtSize(f)
   701→
   702→    if (hasV && hasA) {
   703→      // muxed — instant single-stream
   704→      if (!isStreamable(f)) continue
   705→      instant.push({
   706→        id: f.format_id,
   707→        kind: 'video',
   708→        quality: videoQuality(f),
   709→        container: ext === 'webm' ? 'webm' : 'mp4',
   710→        vcodec: codecLabel(f.vcodec),
   711→        acodec: codecLabel(f.acodec),
   712→        fps: f.fps,
   713→        filesize: size,
   714→        approx: !f.filesize,
   715→        hasAudio: true,
   716→        instant: true,
   717→        needsMerge: false,
   718→        hdr: /hdr/i.test(f.dynamic_range ?? ''),
   719→        recommended: true,
   720→      })
   721→    } else if (hasV) {
   722→      const container = ext === 'webm' ? 'webm' : 'mp4'
   723→      const fmt: PublicFormat = {
   724→        id: f.format_id, // replaced by composite below
   725→        kind: 'video',
   726→        quality: videoQuality(f),
   727→        container,
   728→        vcodec: codecLabel(f.vcodec),
   729→        fps: f.fps,
   730→        filesize: size,
   731→        approx: !f.filesize,
   732→        hasAudio: false,
   733→        instant: false,
   734→        needsMerge: true,
   735→        hdr: /hdr/i.test(f.dynamic_range ?? ''),
   736→      }
   737→      videoOnly.push({ f, fmt })
   738→    } else if (hasA) {
   739→      if (!isStreamable(f)) continue
   740→      audio.push({
   741→        id: f.format_id,
   742→        kind: 'audio',
   743→        quality: audioQuality(f),
   744→        container: ext === 'webm' ? 'webm' : ext === 'm4a' ? 'm4a' : ext,
   745→        acodec: codecLabel(f.acodec),
   746→        filesize: size,
   747→        approx: !f.filesize,
   748→        hasAudio: true,
   749→        instant: true,
   750→        needsMerge: false,
   751→      })
   752→    }
   753→  }
   754→
   755→  // composite merges: pair each video-only track with its best audio partner
   756→  const composites: PublicFormat[] = videoOnly.map(({ f, fmt }) => {
   757→    const a = pickAudio(raw.formats, fmt.container)
   758→    const id = a ? `${f.format_id}|${a.format_id}|${fmt.container}` : f.format_id
   759→    const total = (fmt.filesize ?? 0) + (a ? fmtSize(a) ?? 0 : 0)
   760→    return {
   761→      ...fmt,
   762→      id,
   763→      filesize: total > 0 ? total : fmt.filesize,
   764→      approx: !f.filesize || !a?.filesize,
   765→    }
   766→  })
   767→
   768→  const rankVideo = (f: PublicFormat): number => {
   769→    const h = parseInt(f.quality, 10) || 0
   770→    return h * 100 + (f.fps ?? 0) * 2 + (f.container === 'mp4' ? 1 : 0)
   771→  }
   772→  instant.sort((a, b) => rankVideo(b) - rankVideo(a))
   773→  composites.sort((a, b) => rankVideo(b) - rankVideo(a))
   774→  audio.sort((a, b) => (parseInt(b.quality, 10) || 0) - (parseInt(a.quality, 10) || 0))
   775→
   776→  return {
   777→    infoId,
   778→    title: raw.title,
   779→    author: raw.uploader || raw.channel || '',
   780→    views: raw.view_count ?? 0,
   781→    duration: raw.duration ?? 0,
   782→    thumbnail: thumbProxy(raw.thumbnail),
   783→    webpageUrl: raw.webpage_url,
   784→    source,
   785→    videoFormats: [...composites, ...instant],
   786→    audioFormats: audio,
   787→    mp3Options: hasFfmpeg() ? [320, 192, 128] : [],
   788→  }
   789→}
