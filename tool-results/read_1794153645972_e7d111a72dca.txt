    26→    26→function fmtDuration(sec: number): string {
    27→    27→  if (!sec || sec <= 0) return "";
    28→    28→  const h = Math.floor(sec / 3600);
    29→    29→  const m = Math.floor((sec % 3600) / 60);
    30→    30→  const s = Math.floor(sec % 60);
    31→    31→  return h > 0
    32→    32→    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
    33→    33→    : `${m}:${String(s).padStart(2, "0")}`;
    34→    34→}
    35→    35→
    36→    36→function fmtBytes(b: number): string {
    37→    37→  if (!b || b <= 0) return "";
    38→    38→  const units = ["B", "KB", "MB", "GB"];
    39→    39→  let i = 0;
    40→    40→  let v = b;
    41→    41→  while (v >= 1024 && i < units.length - 1) {
    42→    42→    v /= 1024;
    43→    43→    i++;
    44→    44→  }
    45→    45→  return `${v.toFixed(v >= 100 || i === 0 ? 0 : 1)} ${units[i]}`;
    46→    46→}
    47→    47→
    48→    48→function qualityHeight(q: string): number {
    49→    49→  return parseInt(q, 10) || 0;
    50→    50→}
    51→    51→
    52→    52→function fmtHeight(f: PublicFormat): number {
    53→    53→  const m = f.quality?.match(/(\d{3,4})/);
    54→    54→  return m ? parseInt(m[1], 10) : 0;
    55→    55→}
    56→    56→
    57→    57→function fmtCodec(f: PublicFormat): string | null {
    58→    58→  const v = (f.vcodec ?? "").toLowerCase();
    59→    59→  if (v.startsWith("h264") || v.startsWith("h.264") || v.startsWith("avc")) return "H.264 (AVC)";
    60→    60→  if (v.startsWith("vp9") || v.startsWith("vp09")) return "VP9";
    61→    61→  if (v.startsWith("av1") || v.startsWith("av01")) return "AV1";
    62→    62→  return null;
    63→    63→}

   209→   209→function VideoPanel(p: VideoPanelProps) {
   210→   210→  const { info } = p;
   211→   211→  const vids = info.videoFormats ?? [];
   212→   212→  const auds = info.audioFormats ?? [];
   213→   213→
   214→   214→  /* keep selection valid when a new video arrives */
   215→   215→  useEffect(() => {
   216→   216→    const heights = new Set(vids.map(fmtHeight));
   217→   217→    if (!heights.has(qualityHeight(p.quality))) {
   218→   218→      const best = [...heights].sort((a, b) => b - a).find((h) => h <= 1080) ?? [...heights][0];
   219→   219→      if (best) p.setQuality(`${best}p`);
   220→   220→    }
   221→   221→  }, [info.infoId]);
   222→   222→
   223→   223→  const selHeight = qualityHeight(p.quality);
   224→   224→
   225→   225→  /* ---------- video selection ---------- */
   226→   226→  const atHeight = useMemo(() => vids.filter((f) => fmtHeight(f) === selHeight), [vids, selHeight]);
   227→   227→  const codecAvailable = useMemo(() => {
   228→   228→    const set = new Set<string>();
   229→   229→    for (const f of vids) {
   230→   230→      const c = fmtCodec(f);
   231→   231→      if (c) set.add(c);
   232→   232→    }
   233→   233→    return set;
   234→   234→  }, [vids]);
   235→   235→  const pick = useCallback(
   236→   236→    (wantCodec: string) => {
   237→   237→      const exact = atHeight.find((f) => fmtCodec(f) === wantCodec);
   238→   238→      if (exact) return exact;
   239→   239→      const anyAt = atHeight[0];
   240→   240→      if (anyAt) return anyAt;
   241→   241→      const lower = vids
   242→   242→        .filter((f) => fmtHeight(f) < selHeight && fmtCodec(f) === wantCodec)
   243→   243→        .sort((a, b) => fmtHeight(b) - fmtHeight(a));
   244→   244→      return lower[0] ?? vids.sort((a, b) => fmtHeight(b) - fmtHeight(a))[0] ?? null;
   245→   245→    },
   246→   246→    [atHeight, selHeight, vids],
   247→   247→  );
   248→   248→  const chosen = useMemo(() => pick(p.codec), [pick, p.codec]);
   249→   249→
   250→   250→  /* ---------- audio selection ---------- */
   251→   251→  const audioPick = useCallback(
   252→   252→    (name: string): PublicFormat | null => {
   253→   253→      if (name === "M4A") {
   254→   254→        const native = auds.find((a) => a.container === "m4a");
   255→   255→        return native ?? { id: "convert", kind: "audio", quality: "Audio", container: "m4a", instant: false, needsMerge: false };
   256→   256→      }
   257→   257→      if (name === "Opus") {
   258→   258→        const native = auds.find((a) => a.container === "webm" || a.container === "opus");
   259→   259→        return native ?? { id: "convert", kind: "audio", quality: "Audio", container: "opus", instant: false, needsMerge: false };
   260→   260→      }
   261→   261→      if (name === "MP3")
   262→   262→        return { id: "convert", kind: "audio", quality: "Audio", container: "mp3", instant: false, needsMerge: false };
   263→   263→      if (name === "FLAC")
   264→   264→        return { id: "convert", kind: "audio", quality: "Audio", container: "flac", instant: false, needsMerge: false };
   265→   265→      if (name === "WAV")
   266→   266→        return { id: "convert", kind: "audio", quality: "Audio", container: "wav", instant: false, needsMerge: false };
   267→   267→      return null;
   268→   268→    },
   269→   269→    [auds],
   270→   270→  );
   271→   271→
   272→   272→  const audioChosen = audioPick(p.audioFmt);
   273→   273→  const containers = useMemo(() => {
   274→   274→    const set = new Set<string>();
   275→   275→    for (const f of vids) set.add(f.container);
   276→   276→    if (set.size === 0) set.add("mp4");
   277→   277→    return [...set].join(" · ");
   278→   278→  }, [vids]);
   279→   279→
   280→   280→  const download = () => {
   281→   281→    if (p.audioMode) {
   282→   282→      if (audioChosen) p.onStart(audioChosen, p.audioFmt === "MP3" ? p.mp3Bitrate : undefined);
   283→   283→    } else if (chosen) {
   284→   284→      p.onStart(chosen);
   285→   285→    }
   286→   286→  };
   287→   287→
   288→   288→  const disabled = p.audioMode ? !audioChosen : !chosen;
   289→   289→
   290→   290→  return (
   291→   291→    <section className="card video-card">
   292→   292→      <div className="video-head">
   293→   293→        <div className="thumb">
   294→   294→          <img src={info.thumbnail} alt={info.title} />
   295→   295→        </div>
   296→   296→        <div className="vp-info">
   297→   297→          <div className="vp-title">{info.title}</div>
   298→   298→          <div className="vp-sub">
   299→   299→            {info.author} · {fmtDuration(info.duration)}
   300→   300→            {info.views ? ` · ${info.views.toLocaleString("en-US")} views` : ""}
   301→   301→          </div>
   302→   302→        </div>
   303→   303→      </div>
   304→   304→
   305→   305→      <div className="seg-row">
   306→   306→        <button className={"seg" + (!p.audioMode ? " active" : "")} onClick={() => p.setAudioMode(false)}>
   307→   307→          Video
   308→   308→        </button>
   309→   309→        <button className={"seg" + (p.audioMode ? " active" : "")} onClick={() => p.setAudioMode(true)}>
   310→   310→          Audio
   311→   311→        </button>
   312→   312→      </div>
   313→   313→
   314→   314→      {!p.audioMode ? (
   315→   315→        <>
   316→   316→          <div className="quality-grid">
   317→   317→            {QUALITIES.map((q) => {
   318→   318→              const h = qualityHeight(q);
   319→   319→              const has = vids.some((f) => fmtHeight(f) === h);
   320→   320→              return (
   321→   321→                <button
   322→   322→                  key={q}
   323→   323→                  className={"quality-tile" + (p.quality === q ? " active" : "")}
   324→   324→                  onClick={() => setQualityLocal(p, q)}
   325→   325→                  disabled={!has}
   326→   326→                >
   327→   327→                  {q}
   328→   328→                </button>
   329→   329→              );
   330→   330→            })}
   331→   331→          </div>
   332→   332→          <div className="codec-row">
   333→   333→            {CODECS.map((c) => (
   334→   334→              <button
   335→   335→                key={c}
   336→   336→                className={"chip" + (p.codec === c ? " active" : "")}
   337→   337→                onClick={() => p.setCodec(c)}
   338→   338→                disabled={!codecAvailable.has(c)}
   339→   339→              >
   340→   340→                {c}
   341→   341→              </button>
   342→   342→            ))}
   343→   343→          </div>
   344→   344→        </>
   345→   345→      ) : (
   346→   346→        <>
   347→   347→          <div className="seg-list audio-list">
   348→   348→            {AUDIO.map((a) => (
   349→   349→              <button
   350→   350→                key={a}
   351→   351→                className={"seg" + (p.audioFmt === a ? " active" : "")}
   352→   352→                onClick={() => p.setAudioFmt(a)}
   353→   353→              >
   354→   354→                {a}
   355→   355→              </button>
   356→   356→            ))}
   357→   357→          </div>
   358→   358→          {p.audioFmt === "MP3" && (
   359→   359→            <div className="codec-row">
   360→   360→              {MP3_BITRATES.map((b) => (
   361→   361→                <button
   362→   362→                  key={b}
   363→   363→                  className={"chip" + (p.mp3Bitrate === b ? " active" : "")}
   364→   364→                  onClick={() => p.setMp3Bitrate(b)}
   365→   365→                >
   366→   366→                  {b} kbps
   367→   367→                </button>
   368→   368→              ))}
   369→   369→            </div>
   370→   370→          )}
   371→   371→        </>
   372→   372→      )}
   373→   373→
   374→   374→      <div className="meta-row">
   375→   375→        <span className="muted">Container</span> {containers}
   376→   376→      </div>
   377→   377→
   378→   378→      <div className="dl-row">
   379→   379→        <button className="btn primary big" onClick={download} disabled={disabled} style={{ flex: 1 }}>
   380→   380→          Download
   381→   381→        </button>
   382→   382→        <button
   383→   383→          className="btn ghost big"
   384→   384→          style={{ width: "auto", padding: "0 16px", whiteSpace: "nowrap" }}
   385→   385→          disabled={disabled}
   386→   386→          onClick={() => {
   387→   387→            const fmt = p.audioMode ? audioChosen : chosen;
   388→   388→            if (!fmt) return;
   389→   389→            p.onDirectLink(fmt, p.audioFmt === "MP3" ? p.mp3Bitrate : undefined);
   390→   390→          }}
   391→   391→        >
   392→   392→          🔗 Direct link
   393→   393→        </button>
   394→   394→      </div>
   395→   395→      <div className="hint" style={{ marginTop: 8, textAlign: "center" }}>
   396→   396→        Direct link = the file downloaded to this server first, then you download it from here —
   397→   397→        resumable, no YouTube involved. Works in IDM / aria2 / any browser.
   398→   398→      </div>
   399→   399→    </section>
   400→   400→  );
   401→   401→}

   403→   403→/** quality click also snaps the codec chip to one available at that height */
   404→   404→function setQualityLocal(p: VideoPanelProps, q: string) {
   405→   405→  p.setQuality(q);
   406→   406→  const h = qualityHeight(q);
   407→   407→  const at = p.info.videoFormats.filter((f) => fmtHeight(f) === h);
   408→   408→  if (at.length && !at.some((f) => fmtCodec(f) === p.codec)) {
   409→   409→    const c = fmtCodec(at[0]);
   410→   410→    if (c) p.setCodec(c);
   411→   411→  }
   412→   412→}

   458→   458→function JobRow({
   459→   459→  it,
   460→   460→  onCancel,
   461→   461→  onRemove,
   462→   462→}: {
   463→   463→  it: DownloadItem;
   464→   464→  onCancel: (item: DownloadItem) => void;
   465→   465→  onRemove: (id: string) => void;
   466→   466→}) {
   467→   467→  const working = ["starting", "queued", "downloading", "merging", "saving"].includes(it.state);
   468→   468→  const pct = it.total > 0 ? Math.min(100, Math.round((it.received / it.total) * 100)) : it.state === "done" ? 100 : 0;
   469→   469→  let badge = "";
   470→   470→  let badgeCls = "ok";
   471→   471→  if (it.state === "done") {
   472→   472→    badge = "Ready";
   473→   473→    badgeCls = "ok";
   474→   474→  } else if (it.state === "error") {
   475→   475→    badge = "Failed";
   476→   476→    badgeCls = "err";

   485→   485→    badgeCls = "warn";
   486→   486→  } else {
   487→   487→    badge = `${pct}%`;
   488→   488→    badgeCls = "warn";
   489→   489→  }
   490→   490→
   491→   491→  const sub = [

   495→   495→  ]

   498→   498→
   499→   499→  return (
   500→   500→    <div className="job-row">
   501→   501→      <div style={{ minWidth: 0 }}>
   502→   502→        <div className="job-title">{it.title}</div>

   505→   505→          <button
   506→   506→            className="x-btn"
   507→   507→            style={{ marginTop: 4 }}
   508→   508→            onClick={() => {
   509→   509→              const abs = `${window.location.origin}${it.directLink}`;
   510→   510→              void navigator.clipboard?.writeText(abs).catch(() => undefined);
   511→   511→              if (it.directReady) {
   512→   512→                toast.success("Direct link copied", {
   513→   513→                  description: "The file is stored on the YouWee server — this link downloads it from here with full resume support (IDM, aria2, any browser).",
   514→   514→                });
   515→   515→              } else {

   518→   518→                });
   519→   519→              }
   520→   520→            }}

   525→   525→      </div>
   526→   526→      <div className="job-right">
   527→   527→        <span className={"badge " + badgeCls}>{badge}</span>
   528→   528→        {working ? (
   529→   529→          <button className="x-btn" onClick={() => onCancel(it)}>
   530→   530→            Cancel
   531→   531→          </button>
   532→   532→        ) : (
   533→   533→          <button className="x-btn" onClick={() => onRemove(it.id)}>
   534→   534→            ✕
   535→   535→          </button>

   537→   537→      </div>
   538→   538→      {working && (
   539→   539→        <div className="job-prog">

   541→   541→            <div className="progress inner" style={{ width: `${Math.max(2, pct)}%` }} />
   542→   542→          </div>
   543→   543→        </div>
   544→   544→      )}
   545→   545→    </div>
   546→   546→  );
   547→   547→}
