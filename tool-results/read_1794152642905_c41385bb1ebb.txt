    63→    63→/**
    64→    64→ * Honest status ladder for the extraction poll. The server cycles through
    65→    65→ * yt-dlp → Piped/Invidious mirrors → the community-mirror backend on every
    66→    66→ * attempt; the copy reflects what actually happens at each stage instead of
    67→    67→ * a vague "busy" line.

    69→    69→function pendingStatus(attempts: number, elapsedMs: number): string {
    70→    70→  if (attempts <= 1) return "Connecting to YouTube…";
    71→    71→  if (attempts <= 3) return `YouTube is throttling this route — retrying (attempt ${attempts})…`;
    72→    72→  if (elapsedMs > 180_000)
    73→    73→    return `Still working — the mirror farm is preparing your file (attempt ${attempts})…`;
    74→    74→  return `YouTube blocked us — community mirrors engaged (attempt ${attempts})…`;
    75→    75→}

    81→    81→type SavePicker = (opts: {
    82→    82→  suggestedName: string;
    83→    83→  types: Array<{ description: string; accept: Record<string, string[]> }>;
    84→    84→}) => Promise<FileSystemFileHandle>;
    85→    85→
    86→    86→async function openWritable(
    87→    87→  filename: string,
    88→    88→  mime: string,
    89→    89→  ext: string,
    90→    90→): Promise<FileSystemWritableFileStream | null> {
    91→    91→  const w = window as unknown as { showSaveFilePicker?: SavePicker };
    92→    92→  // automation/headless environments can't render the native picker → blob path
    93→    93→  const automated = typeof navigator !== "undefined" && navigator.webdriver === true;
    94→    94→  if (!w.showSaveFilePicker || automated) return null;
    95→    95→  try {
    96→    96→    const handle = await w.showSaveFilePicker({
    97→    97→      suggestedName: filename,
    98→    98→      types: [{ description: `${ext.toUpperCase()} file`, accept: { [mime]: [`.${ext}`] } }],
    99→    99→    });
   100→   100→    return await handle.createWritable();
   101→   101→  } catch (e) {
   102→   102→    const name = (e as DOMException)?.name;
   103→   103→    if (name === "AbortError") throw e; // user cancelled → abort whole download
   104→   104→    return null; // permission/iframe issue → blob fallback
   105→   105→  }
   106→   106→}

   108→   108→function mimeForExt(ext: string): string {
   109→   109→  switch (ext) {
   110→   110→    case "mp4":
   111→   111→      return "video/mp4";
   112→   112→    case "webm":
   113→   113→      return "video/webm";
   114→   114→    case "mkv":
   115→   115→      return "video/x-matroska";
   116→   116→    case "m4a":
   117→   117→      return "audio/mp4";
   118→   118→    case "mp3":
   119→   119→      return "audio/mpeg";
   120→   120→    default:
   121→   121→      return "application/octet-stream";
   122→   122→  }
   123→   123→}

   125→   125→/**
   126→   126→ * Build the /api/dl resolve link for a selection (pre-store fallback).
   127→   127→ * When a download manager (or browser) hits it, the SERVER downloads the
   128→   128→ * file fully first (answering 503 + Retry-After meanwhile), then serves the
   129→   129→ * finished file with Range/resume — the client never touches YouTube.
   130→   130→ */
   131→   131→export function buildDlLink(
   132→   132→  fmt: PublicFormat,
   133→   133→  info: PublicInfo,
   134→   134→  bitrate?: number,
   135→   135→): string {
   136→   136→  const isConversion =
   137→   137→    fmt.kind === "audio" &&
   138→   138→    (fmt.id === "convert" || ["mp3", "flac", "wav"].includes(fmt.container));
   139→   139→  const f = isConversion ? fmt.container : fmt.id;
   140→   140→  const qs = new URLSearchParams({
   141→   141→    u: info.webpageUrl,
   142→   142→    f,
   143→   143→    q: fmt.quality || "",
   144→   144→    t: info.title || "",
   145→   145→  });
   146→   146→  if (isConversion && fmt.container === "mp3") qs.set("b", String(bitrate ?? 320));
   147→   147→  return `/api/dl?${qs.toString()}`;
   148→   148→}

   154→   154→export function useDownloader() {
   155→   155→  const { t } = useLang();
   156→   156→
   157→   157→  const [url, setUrl] = useState("");
   158→   158→  const [phase, setPhase] = useState<"idle" | "loading" | "ready" | "error">("idle");
   159→   159→  const [pendingNote, setPendingNote] = useState<string | null>(null);
   160→   160→  const [info, setInfo] = useState<PublicInfo | null>(null);
   161→   161→  const [errorMsg, setErrorMsg] = useState<string | null>(null);
   162→   162→  const [errorCode, setErrorCode] = useState<string | null>(null);
   163→   163→  const [items, setItems] = useState<DownloadItem[]>([]);
   164→   164→  const [history, setHistory] = useState<HistoryEntry[]>([]);
   165→   165→
   166→   166→  const controllers = useRef(new Map<string, AbortController>());
   167→   167→  const samplers = useRef(
   168→   168→    new Map<string, { lastT: number; lastB: number; ema: number; lastRender: number }>(),
   169→   169→  );
   170→   170→
   171→   171→  // Load persisted history AFTER hydration — reading localStorage during render
   172→   172→  // makes the first client render differ from the server HTML (hydration error).
   173→   173→  useEffect(() => {
   174→   174→    try {
   175→   175→      const raw = window.localStorage.getItem(HISTORY_KEY);
   176→   176→      if (raw) setHistory(JSON.parse(raw) as HistoryEntry[]);
   177→   177→    } catch {
   178→   178→      /* ignore */
   179→   179→    }
   180→   180→  }, []);
   181→   181→
   182→   182→  const pushHistory = useCallback((entry: HistoryEntry) => {
   183→   183→    setHistory((prev) => {
   184→   184→      const next = [entry, ...prev].slice(0, 20);
   185→   185→      try {
   186→   186→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next));
   187→   187→      } catch {
   188→   188→        /* ignore */
   189→   189→      }
   190→   190→      return next;
   191→   191→    });
   192→   192→  }, []);
   193→   193→
   194→   194→  const clearHistory = useCallback(() => {
   195→   195→    setHistory([]);
   196→   196→    try {
   197→   197→      window.localStorage.removeItem(HISTORY_KEY);
   198→   198→    } catch {
   199→   199→      /* ignore */
   200→   200→    }
   201→   201→  }, []);
   202→   202→
   203→   203→  const patchItem = useCallback((id: string, patch: Partial<DownloadItem>, force = false) => {
   204→   204→    setItems((prev) => {
   205→   205→      const idx = prev.findIndex((it) => it.id === id);
   206→   206→      if (idx === -1) return prev;
   207→   207→      const next = [...prev];
   208→   208→      next[idx] = { ...next[idx], ...patch };
   209→   209→      return next;
   210→   210→    });
   211→   211→    void force;
   212→   212→  }, []);
   213→   213→
   214→   214→  /* ---------------- info fetching ---------------- */
   215→   215→
   216→   216→  const fetchInfo = useCallback(
   217→   217→    async (rawUrl: string) => {
   218→   218→      const clean = rawUrl.trim();
   219→   219→      if (!clean) return;
   220→   220→      setPhase("loading");
   221→   221→      setErrorMsg(null);
   222→   222→      setErrorCode(null);
   223→   223→      try {
   224→   224→        const res = await fetch("/api/info", {
   225→   225→          method: "POST",
   226→   226→          headers: { "Content-Type": "application/json" },
   227→   227→          body: JSON.stringify({ url: clean }),
   228→   228→          // the server answers within ~26 s: either the result or a 202
   229→   229→          // "pending" that its background retry loop now owns
   230→   230→          signal: AbortSignal.timeout(30_000),
   231→   231→        });
   232→   232→        if (res.status === 202) {
   233→   233→          // cobalt-style session: the server keeps retrying through tunnel
   234→   234→          // phases — poll until done / failed / user gives up
   235→   235→          const deadline = Date.now() + 12 * 60_000;
   236→   236→          while (Date.now() < deadline) {
   237→   237→            await new Promise((r) => setTimeout(r, 4_000));
   238→   238→            const pr = await fetch(`/api/info?u=${encodeURIComponent(clean)}`, {
   239→   239→              signal: AbortSignal.timeout(10_000),
   240→   240→            }).catch(() => null);
   241→   241→            if (!pr) {
   242→   242→              setPendingNote("Reconnecting…");
   243→   243→              continue;
   244→   244→            }
   245→   245→            if (pr.status === 200) {
   246→   246→              const json = await pr.json();
   247→   247→              if (json?.info) {
   248→   248→                setInfo(json.info as PublicInfo);
   249→   249→                setPhase("ready");
   250→   250→                setPendingNote(null);
   251→   251→                return;
   252→   252→              }
   253→   253→              continue;
   254→   254→            }
   255→   255→            if (pr.status === 202) {
   256→   256→              const j = await pr.json().catch(() => null);
   257→   257→              const n = j?.attempts ?? 0;
   258→   258→              const elapsed = j?.elapsedMs ?? 0;
   259→   259→              setPendingNote(pendingStatus(n, elapsed));
   260→   260→              continue;
   261→   261→            }
   262→   262→            const j = await pr.json().catch(() => null);
   263→   263→            setPhase("error");
   264→   264→            setPendingNote(null);
   265→   265→            setErrorCode(j?.error?.code || "FETCH_FAILED");
   266→   266→            setErrorMsg(j?.error?.message || "Failed");
   267→   267→            return;
   268→   268→          }
   269→   269→          setPhase("error");
   270→   270→          setPendingNote(null);
   271→   271→          setErrorCode("TIMEOUT");
   272→   272→          setErrorMsg("The engine could not reach YouTube in time — try again in a few minutes.");
   273→   273→          return;
   274→   274→        }
   275→   275→        const json = await res.json();
   276→   276→        if (!res.ok || json.error) {
   277→   277→          setPhase("error");
   278→   278→          setPendingNote(null);
   279→   279→          setErrorCode(json?.error?.code || "FETCH_FAILED");
   280→   280→          setErrorMsg(json?.error?.message || "Failed");
   281→   281→          return;
   282→   282→        }
   283→   283→        setInfo(json.info as PublicInfo);
   284→   284→        setPendingNote(null);
   285→   285→        setPhase("ready");
   286→   286→      } catch (e) {
   287→   287→        setPhase("error");
   288→   288→        setPendingNote(null);
   289→   289→        setErrorCode("FETCH_FAILED");
   290→   290→        const aborted = e instanceof DOMException && e.name === "TimeoutError";
   291→   291→        setErrorMsg(
   292→   292→          aborted
   293→   293→            ? "The engine took too long — the tunnel is congested. Try again in a minute."
   294→   294→            : "Network error",
   295→   295→        );
   296→   296→      }
   297→   297→    },
   298→   298→    [],
   299→   299→  );
   300→   300→
   301→   301→  const reset = useCallback(() => {
   302→   302→    setUrl("");
   303→   303→    setInfo(null);
   304→   304→    setPhase("idle");
   305→   305→    setErrorMsg(null);
   306→   306→    setErrorCode(null);
   307→   307→    setPendingNote(null);
   308→   308→  }, []);
   309→   309→
   310→   310→  /* ---------------- sampler ---------------- */
   311→   311→
   312→   312→  const sampleSpeed = useCallback((id: string, received: number): number => {
   313→   313→    const now = performance.now();
   314→   314→    const s = samplers.current.get(id) ?? { lastT: now, lastB: received, ema: 0, lastRender: 0 };
   315→   315→    const dt = (now - s.lastT) / 1000;
   316→   316→    if (dt >= 0.4) {
   317→   317→      const inst = (received - s.lastB) / dt;
   318→   318→      s.ema = s.ema === 0 ? inst : s.ema * 0.7 + inst * 0.3;
   319→   319→      s.lastT = now;
   320→   320→      s.lastB = received;
   321→   321→    }
   322→   322→    samplers.current.set(id, s);
   323→   323→    return s.ema;
   324→   324→  }, []);
   325→   325→
   326→   326→  /* ---------------- core streaming save (auto-resume) ---------------- */
   327→   327→
   328→   328→  const streamToDisk = useCallback(
   329→   329→    async (
   330→   330→      itemId: string,
   331→   331→      open: (range?: string) => Promise<Response>,
   332→   332→      item: {
   333→   333→        filename: string;
   334→   334→        ext: string;
   335→   335→        label: string;
   336→   336→        title: string;
   337→   337→        videoUrl?: string;
   338→   338→        thumb?: string;
   339→   339→        directUrl?: string; // native-download fallback for browsers without the File System Access API
   340→   340→        storedDirectUrl?: string; // stable /api/file/<token> link once the file is stored server-side
   341→   341→      },
   342→   342→      expectedTotal: number,
   343→   343→      signal: AbortSignal,
   344→   344→    ) => {
   345→   345→      // Slow/flaky networks: a dropped connection or a hiccup on the server no
   346→   346→      // longer kills the transfer — we re-request with `Range: bytes=<n>-` and
   347→   347→      // continue into the SAME file, up to 15 times.
   348→   348→      const MAX_RESUMES = 15;
   349→   349→      let received = 0; // absolute offset across resumes
   350→   350→      let total = expectedTotal || 0;
   351→   351→      let writer: FileSystemWritableFileStream | null = null;
   352→   352→      let writerReady = false;
   353→   353→      const chunks: Uint8Array[] = []; // blob fallback path
   354→   354→      let resumes = 0;
   355→   355→      const mime = mimeForExt(item.ext);
   356→   356→      const backoff = () => new Promise((r) => setTimeout(r, 700 + resumes * 500));
   357→   357→
   358→   358→      for (;;) {
   359→   359→        /* ---- (re)open the response ---- */
   360→   360→        let res: Response;
   361→   361→        try {
   362→   362→          res = await open(received > 0 ? `bytes=${received}-` : undefined);
   363→   363→        } catch {
   364→   364→          if (signal.aborted) throw new DOMException("Aborted", "AbortError");
   365→   365→          if (received === 0 || resumes >= MAX_RESUMES) throw new Error("FETCH_FAILED");
   366→   366→          resumes++;
   367→   367→          await backoff();
   368→   368→          continue;
   369→   369→        }
   370→   370→
   371→   371→        if (!res.ok && res.status !== 206) {
   372→   372→          let code = "FETCH_FAILED";
   373→   373→          try {
   374→   374→            const j = await res.json();
   375→   375→            code = j?.error?.code || code;
   376→   376→          } catch {
   377→   377→            /* ignore */
   378→   378→          }
   379→   379→          // fatal: nothing to resume from, or the source is gone for good
   380→   380→          if (
   381→   381→            received === 0 ||
   382→   382→            code === "NOT_FOUND" ||
   383→   383→            code === "EXPIRED" ||
   384→   384→            code === "USE_JOB" ||
   385→   385→            code === "NO_FORMATS" ||
   386→   386→            res.status === 416
   387→   387→          ) {
   388→   388→            throw new Error(code);
   389→   389→          }
   390→   390→          if (resumes >= MAX_RESUMES) throw new Error(code);
   391→   391→          resumes++;
   392→   392→          await backoff();
   393→   393→          continue;
   394→   394→        }
   395→   395→
   396→   396→        if (!res.body) {
   397→   397→          if (received === 0) throw new Error("FETCH_FAILED");
   398→   398→          resumes++;
   399→   399→          await backoff();
   400→   400→          continue;
   401→   401→        }
   402→   402→
   403→   403→        /* ---- size bookkeeping ---- */
   404→   404→        const clen = Number(res.headers.get("content-length")) || 0;
   405→   405→        const crange = res.headers.get("content-range");
   406→   406→        if (crange) {
   407→   407→          const m = crange.match(/\/(\d+)/);
   408→   408→          if (m) total = parseInt(m[1], 10);
   409→   409→        } else if (received === 0 && clen > 0) {
   410→   410→          total = clen;
   411→   411→        }
   412→   412→
   413→   413→        if (!writerReady) {
   414→   414→          writerReady = true;
   415→   415→          try {
   416→   416→            writer = await openWritable(item.filename, mime, item.ext);
   417→   417→          } catch {
   418→   418→            // user dismissed the save dialog → cancel silently
   419→   419→            patchItem(itemId, { state: "cancelled" });
   420→   420→            return;
   421→   421→          }
   422→   422→        }
   423→   423→
   424→   424→        patchItem(itemId, { state: "downloading", total, received });
   425→   425→
   426→   426→        /* ---- pump the current stream ---- */
   427→   427→        const reader = res.body.getReader();
   428→   428→        let streamError: unknown = null;
   429→   429→        let lastUi = 0;
   430→   430→        try {
   431→   431→          for (;;) {
   432→   432→            if (signal.aborted) throw new DOMException("Aborted", "AbortError");
   433→   433→            const { done, value } = await reader.read();
   434→   434→            if (done) break;
   435→   435→            received += value.length;
   436→   436→            if (writer) {
   437→   437→              await writer.write(value);
   438→   438→            } else {
   439→   439→              chunks.push(value);
   440→   440→            }
   441→   441→            const speed = sampleSpeed(itemId, received);
   442→   442→            const now = performance.now();
   443→   443→            if (now - lastUi > 140) {
   444→   444→              lastUi = now;
   445→   445→              patchItem(itemId, { received, speed, total: total || Math.max(total, received) });
   446→   446→            }
   447→   447→          }
   448→   448→        } catch (e) {
   449→   449→          streamError = e;
   450→   450→        } finally {
   451→   451→          try {
   452→   452→            reader.releaseLock();
   453→   453→          } catch {
   454→   454→            /* ignore */
   455→   455→          }
   456→   456→          try {
   457→   457→            await reader.cancel();
   458→   458→          } catch {
   459→   459→            /* ignore */
   460→   460→          }
   461→   461→        }
   462→   462→
   463→   463→        if (streamError && signal.aborted) {
   464→   464→          try {
   465→   465→            await writer?.abort();
   466→   466→          } catch {
   467→   467→            /* ignore */
   468→   468→          }
   469→   469→          throw new DOMException("Aborted", "AbortError");
   470→   470→        }
   471→   471→
   472→   472→        // clean EOF with the full size delivered → done; clean EOF but short →
   473→   473→        // treat exactly like a broken stream and resume
   474→   474→        const complete = !streamError && (total === 0 || received >= total);
   475→   475→        if (complete) break;
   476→   476→
   477→   477→        resumes++;
   478→   478→        if (resumes > MAX_RESUMES) {
   479→   479→          try {
   480→   480→            await writer?.abort();
   481→   481→          } catch {
   482→   482→            /* ignore */
   483→   483→          }
   484→   484→          throw (streamError as Error) || new Error("INCOMPLETE");
   485→   485→        }
   486→   486→        console.log(`[download] auto-resume at ${received}/${total || "?"} (attempt ${resumes})`);
   487→   487→        patchItem(itemId, { speed: 0 });
   488→   488→        await backoff();
   489→   489→      }
   490→   490→
   491→   491→      if (writer) {
   492→   492→        await writer.close();
   493→   493→      } else if (item.directUrl) {
   494→   494→        // No File System Access API (Firefox/Safari) — do NOT buffer the file
   495→   495→        // in RAM as a blob. Hand the transfer to the browser's native download
   496→   496→        // manager via Content-Disposition (streams to disk, supports pause on
   497→   497→        // most browsers, zero memory blowup on multi-hundred-MB files).
   498→   498→        patchItem(itemId, { state: "saving" }, true);
   499→   499→        const a = document.createElement("a");
   500→   500→        a.href = item.directUrl;
   501→   501→        a.download = item.filename;
   502→   502→        document.body.appendChild(a);
   503→   503→        a.click();
   504→   504→        a.remove();
   505→   505→      } else {
   506→   506→        patchItem(itemId, { state: "saving" }, true);
   507→   507→        const blob = new Blob(chunks as BlobPart[], { type: mime });
   508→   508→        const objUrl = URL.createObjectURL(blob);
   509→   509→        const a = document.createElement("a");
   510→   510→        a.href = objUrl;
   511→   511→        a.download = item.filename;
   512→   512→        document.body.appendChild(a);
   513→   513→        a.click();
   514→   514→        a.remove();
   515→   515→        setTimeout(() => URL.revokeObjectURL(objUrl), 60_000);
   516→   516→      }
   517→   517→
   518→   518→      patchItem(itemId, { state: "done", received, total: total || received, speed: 0 }, true);
   519→   519→      pushHistory({
   520→   520→        id: `${Date.now()}-${item.label}`,
   521→   521→        title: item.title,
   522→   522→        thumb: item.thumb || "",
   523→   523→        label: item.label,
   524→   524→        ext: item.ext,
   525→   525→        size: received,
   526→   526→        at: Date.now(),
   527→   527→        videoUrl: item.videoUrl || "",
   528→   528→        directUrl: item.storedDirectUrl,
   529→   529→      });
   530→   530→      toast.success(t("downloadComplete"), { description: `${item.filename} · ${formatBytes(received)}` });
   531→   531→    },
   532→   532→    [patchItem, pushHistory, sampleSpeed, t],
   533→   533→  );
   534→   534→
   535→   535→  /* ---------------- job polling ---------------- */
   536→   536→
   537→   537→  interface JobPoll {
   538→   538→    state: string;
   539→   539→    phase: string;
   540→   540→    progress: number;
   541→   541→    speed: string | null;
   542→   542→    receivedBytes: number;
   543→   543→    sizeBytes: number | null;
   544→   544→    error: string | null;
   545→   545→    directUrl?: string | null;
   546→   546→  }
   547→   547→
   548→   548→  const pollJob = useCallback(
   549→   549→    async (itemId: string, jobId: string, signal: AbortSignal): Promise<JobPoll> => {
   550→   550→      for (;;) {
   551→   551→        if (signal.aborted) throw new DOMException("Aborted", "AbortError");
   552→   552→        await new Promise((r) => setTimeout(r, 700));
   553→   553→        const res = await fetch(`/api/jobs/${jobId}`, { signal });
   554→   554→        const json = await res.json();
   555→   555→        if (!res.ok || json.error) throw new Error(json?.error?.code || "NOT_FOUND");
   556→   556→        let job = json.job as JobPoll;
   557→   557→        if (job.state === "ready") {
   558→   558→          // the finished file is registered in the store right after "ready"
   559→   559→          // — one extra poll picks up the stable token link if it raced us
   560→   560→          if (!job.directUrl) {
   561→   561→            await new Promise((r) => setTimeout(r, 900));
   562→   562→            try {
   563→   563→              const again = await fetch(`/api/jobs/${jobId}`, { signal });
   564→   564→              const j2 = await again.json();
   565→   565→              if (again.ok && j2?.job?.directUrl) job = j2.job as JobPoll;
   566→   566→            } catch {
   567→   567→              /* keep what we have */
   568→   568→            }
   569→   569→          }
   570→   570→          return job;
   571→   571→        }
   572→   572→        if (job.state === "error") throw new Error(job.error || "INTERNAL");
   573→   573→        if (job.state === "cancelled") throw new DOMException("Cancelled", "AbortError");
   574→   574→        patchItem(itemId, {
   575→   575→          state: job.state === "merging" ? "merging" : job.state === "downloading" ? "downloading" : "queued",
   576→   576→          received: job.receivedBytes,
   577→   577→          total: job.sizeBytes ?? 0,
   578→   578→        });
   579→   579→      }
   580→   580→    },
   581→   581→    [patchItem],
   582→   582→  );
   583→   583→
   584→   584→  /* ---------------- start downloads ---------------- */
   585→   585→
   586→   586→  const startDownload = useCallback(
   587→   587→    async (fmt: PublicFormat, infoData: PublicInfo, overrideBitrate?: number) => {
   588→   588→      // audio conversions: mp3/flac/wav always re-encode; m4a/opus only when no
   589→   589→      // native stream of that container exists (fmt.id === "convert")
   590→   590→      const isConversion =
   591→   591→        fmt.kind === "audio" &&
   592→   592→        (fmt.id === "convert" || ["mp3", "flac", "wav"].includes(fmt.container));
   593→   593→      const label = isConversion
   594→   594→        ? fmt.container === "mp3"
   595→   595→          ? `MP3 ${overrideBitrate ?? 320}`
   596→   596→          : fmt.container.toUpperCase()
   597→   597→        : fmt.kind === "video"
   598→   598→          ? `${fmt.quality} · ${fmt.container.toUpperCase()}`
   599→   599→          : `${fmt.quality} · ${fmt.container.toUpperCase()}`;
   600→   600→      const id = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   601→   601→      const filename = `${infoData.title.slice(0, 80).replace(/[\\/:*?"<>|]/g, " ").trim()} (${label.replace(/[\\/:*?"<>|]/g, "-")}).${fmt.container}`;
   602→   602→      const ctrl = new AbortController();
   603→   603→      controllers.current.set(id, ctrl);
   604→   604→      samplers.current.set(id, { lastT: performance.now(), lastB: 0, ema: 0, lastRender: 0 });
   605→   605→
   606→   606→      const shell: DownloadItem = {
   607→   607→        id,
   608→   608→        title: infoData.title,
   609→   609→        label,
   610→   610→        filename,
   611→   611→        ext: fmt.container,
   612→   612→        state: "starting",
   613→   613→        received: 0,
   614→   614→        total: fmt.filesize || 0,
   615→   615→        speed: 0,
   616→   616→        videoUrl: infoData.webpageUrl,
   617→   617→        thumb: infoData.thumbnail,
   618→   618→        directLink: buildDlLink(fmt, infoData, overrideBitrate),
   619→   619→      };
   620→   620→      setItems((prev) => [shell, ...prev].slice(0, 12));
   621→   621→
   622→   622→      try {
   623→   623→        /* ================================================================
   624→   624→         * MIDDLEMAN FLOW — every download is a server job. The server
   625→   625→         * (fast pipe + WARP) pulls from YouTube FIRST; the user's browser
   626→   626→         * or external manager then fetches the finished file from THIS
   627→   627→         * app with Range/resume. The user's own connection never touches
   628→   628→         * YouTube, which is what makes this reliable on slow/censored
   629→   629→         * networks (Iran etc.).
   630→   630→         * ================================================================ */
   631→   631→        {
   632→   632→          const isMerge = fmt.needsMerge;
   633→   633→          const res = await fetch("/api/jobs", {
   634→   634→            method: "POST",
   635→   635→            headers: { "Content-Type": "application/json" },
   636→   636→            body: JSON.stringify({
   637→   637→              infoId: infoData.infoId,
   638→   638→              mode: isConversion ? "mp3" : isMerge ? "merge" : "file",
   639→   639→              videoFormatId: isConversion ? undefined : fmt.id,
   640→   640→              audioFormat: isConversion ? fmt.container : undefined,
   641→   641→              bitrate: isConversion ? overrideBitrate ?? 320 : undefined,
   642→   642→              title: infoData.title,
   643→   643→              webpageUrl: infoData.webpageUrl,
   644→   644→              quality: fmt.quality || "",
   645→   645→            }),
   646→   646→            signal: ctrl.signal,
   647→   647→          });
   648→   648→          const json = await res.json();
   649→   649→          if (!res.ok || json.error) throw new Error(json?.error?.code || "INTERNAL");
   650→   650→          const job = json.job as { id: string; sizeBytes: number | null };
   651→   651→          patchItem(id, { jobId: job.id, state: "queued", total: job.sizeBytes || fmt.filesize || 0 }, true);
   652→   652→          const done = await pollJob(id, job.id, ctrl.signal);
   653→   653→          // the file is now STORED server-side → swap in the stable token link
   654→   654→          if (done.directUrl) {
   655→   655→            patchItem(id, { directLink: done.directUrl, directReady: true }, true);
   656→   656→          }
   657→   657→          const fileUrl = `/api/jobs/${job.id}/file`;
   658→   658→          await streamToDisk(
   659→   659→            id,
   660→   660→            (range) =>
   661→   661→              fetch(fileUrl, {
   662→   662→                signal: ctrl.signal,
   663→   663→                headers: range ? { Range: range } : undefined,
   664→   664→              }),
   665→   665→            { ...shell, directUrl: fileUrl, storedDirectUrl: done.directUrl ?? undefined },
   666→   666→            fmt.filesize || 0,
   667→   667→            ctrl.signal,
   668→   668→          );
   669→   669→        }
   670→   670→      } catch (e) {
   671→   671→        const err = e as Error & { name?: string };
   672→   672→        if (err.name === "AbortError" || err.message === "Cancelled") {
   673→   673→          patchItem(id, { state: "cancelled", speed: 0 }, true);
   674→   674→        } else {
   675→   675→          patchItem(id, { state: "error", error: err.message || "INTERNAL", speed: 0 }, true);
   676→   676→          toast.error(t("downloadFailed"), { description: t(errorKey(err.message)) });
   677→   677→        }
   678→   678→      } finally {
   679→   679→        controllers.current.delete(id);
   680→   680→        samplers.current.delete(id);
   681→   681→      }
   682→   682→    },
   683→   683→    [patchItem, pollJob, streamToDisk, t],
   684→   684→  );
   685→   685→
   686→   686→  const cancelItem = useCallback(
   687→   687→    (item: DownloadItem) => {
   688→   688→      const ctrl = controllers.current.get(item.id);
   689→   689→      ctrl?.abort();
   690→   690→      controllers.current.delete(item.id);
   691→   691→      if (item.jobId) {
   692→   692→        void fetch(`/api/jobs/${item.jobId}`, { method: "DELETE" }).catch(() => undefined);
   693→   693→      }
   694→   694→      patchItem(item.id, { state: "cancelled", speed: 0 }, true);
   695→   695→    },
   696→   696→    [patchItem],
   697→   697→  );
   698→   698→
   699→   699→  const removeItem = useCallback(
   700→   700→    (id: string) => {
   701→   701→      setItems((prev) => prev.filter((it) => it.id !== id));
   702→   702→    },
   703→   703→    [],
   704→   704→  );
   705→   705→
   706→   706→  const clearFinished = useCallback(() => {
   707→   707→    setItems((prev) => prev.filter((it) => it.state !== "done" && it.state !== "cancelled" && it.state !== "error"));
   708→   708→  }, []);
   709→   709→
   710→   710→  /* ---------------- direct link (fetch-first) ---------------- */
   711→   711→
   712→   712→  /**
   713→   713→   * "🔗 Direct link" — the file must be ON THE SERVER before a link is
   714→   714→   * handed out. Ready file → copy the stable /api/file/<token> link
   715→   715→   * instantly. Otherwise the server starts downloading it first and a
   716→   716→   * tracking row appears in Downloads; the stable link shows up on that
   717→   717→   * row (Copy direct link) the moment the file is stored.
   718→   718→   */
   719→   719→  const getDirectLink = useCallback(
   720→   720→    async (fmt: PublicFormat, infoData: PublicInfo, overrideBitrate?: number) => {
   721→   721→      const isConversion =
   722→   722→        fmt.kind === "audio" &&
   723→   723→        (fmt.id === "convert" || ["mp3", "flac", "wav"].includes(fmt.container));
   724→   724→      const spec = {
   725→   725→        url: infoData.webpageUrl,
   726→   726→        format: isConversion ? fmt.container : fmt.id,
   727→   727→        quality: fmt.quality || "",
   728→   728→        bitrate: isConversion && fmt.container === "mp3" ? overrideBitrate ?? 320 : undefined,
   729→   729→        title: infoData.title,
   730→   730→      };
   731→   731→      try {
   732→   732→        const res = await fetch("/api/dl-resolve", {
   733→   733→          method: "POST",
   734→   734→          headers: { "Content-Type": "application/json" },
   735→   735→          body: JSON.stringify(spec),
   736→   736→        });
   737→   737→        const json = await res.json();
   738→   738→        if (json.status === "ready" && json.directUrl) {
   739→   739→          const abs = `${window.location.origin}${json.directUrl}`;
   740→   740→          void navigator.clipboard?.writeText(abs).catch(() => undefined);
   741→   741→          toast.success("Direct link copied", {
   742→   742→            description: `Stored on the YouWee server (${formatBytes(json.sizeBytes || 0)}) — downloads from here with full resume support. Paste it in IDM, aria2 or any browser.`,
   743→   743→          });
   744→   744→          return;
   745→   745→        }
   746→   746→        if (json.status === "busy") {
   747→   747→          const label = isConversion
   748→   748→            ? fmt.container.toUpperCase()
   749→   749→            : fmt.kind === "video"
   750→   750→              ? `${fmt.quality} · ${fmt.container.toUpperCase()}`
   751→   751→              : `${fmt.quality} · ${fmt.container.toUpperCase()}`;
   752→   752→          const id = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   753→   753→          const ctrl = new AbortController();
   754→   754→          controllers.current.set(id, ctrl);
   755→   755→          setItems((prev) =>
   756→   756→            [
   757→   757→              {
   758→   758→                id,
   759→   759→                title: infoData.title,
   760→   760→                label,
   761→   761→                filename: `${infoData.title.slice(0, 80)} (${label}).${fmt.container}`,
   762→   762→                ext: fmt.container,
   763→   763→                state: "queued" as ItemState,
   764→   764→                received: 0,
   765→   765→                total: fmt.filesize || 0,
   766→   766→                speed: 0,
   767→   767→                jobId: json.jobId || undefined,
   768→   768→                videoUrl: infoData.webpageUrl,
   769→   769→                thumb: infoData.thumbnail,
   770→   770→                directLink: buildDlLink(fmt, infoData, overrideBitrate),
   771→   771→              },
   772→   772→              ...prev,
   773→   773→            ].slice(0, 12),
   774→   774→          );
   775→   775→          toast.info("Fetching on the server first…", {
   776→   776→            description: "The file is being downloaded to YouWee's server at full speed. The direct link appears on the row below as soon as it's stored.",
   777→   777→          });
   778→   778→          try {
   779→   779→            if (json.jobId) {
   780→   780→              const done = await pollJob(id, json.jobId, ctrl.signal);
   781→   781→              if (done.directUrl) {
   782→   782→                patchItem(id, { state: "done", directLink: done.directUrl, directReady: true, total: done.sizeBytes ?? 0 }, true);
   783→   783→                pushHistory({
   784→   784→                  id: `${Date.now()}-${label}`,
   785→   785→                  title: infoData.title,
   786→   786→                  thumb: infoData.thumbnail || "",
   787→   787→                  label,
   788→   788→                  ext: fmt.container,
   789→   789→                  size: done.sizeBytes ?? 0,
   790→   790→                  at: Date.now(),
   791→   791→                  videoUrl: infoData.webpageUrl,
   792→   792→                  directUrl: done.directUrl,
   793→   793→                });
   794→   794→                toast.success("Direct link is ready", {
   795→   795→                  description: "The file is stored on the server — grab it from the row below (Copy direct link).",
   796→   796→                });
   797→   797→              } else {
   798→   798→                patchItem(id, { state: "done", total: done.sizeBytes ?? 0 }, true);
   799→   799→              }
   800→   800→            } else {
   801→   801→              // warm-up still resolving (no job yet) → poll the resolver
   802→   802→              for (;;) {
   803→   803→                if (ctrl.signal.aborted) throw new DOMException("Aborted", "AbortError");
   804→   804→                await new Promise((r) => setTimeout(r, 2500));
   805→   805→                const r2 = await fetch("/api/dl-resolve", {
   806→   806→                  method: "POST",
   807→   807→                  headers: { "Content-Type": "application/json" },
   808→   808→                  body: JSON.stringify(spec),
   809→   809→                  signal: ctrl.signal,
   810→   810→                });
   811→   811→                const j2 = await r2.json();
   812→   812→                if (j2.status === "ready" && j2.directUrl) {
   813→   813→                  patchItem(id, { state: "done", directLink: j2.directUrl, directReady: true, total: j2.sizeBytes ?? 0 }, true);
   814→   814→                  toast.success("Direct link is ready", {
   815→   815→                    description: "The file is stored on the server — grab it from the row below (Copy direct link).",
   816→   816→                  });
   817→   817→                  break;
   818→   818→                }
   819→   819→                if (j2.status === "failed") throw new Error(j2.code || "INTERNAL");
   820→   820→              }
   821→   821→            }
   822→   822→          } catch (e) {
   823→   823→            const err = e as Error & { name?: string };
   824→   824→            if (err.name === "AbortError") patchItem(id, { state: "cancelled" }, true);
   825→   825→            else {
   826→   826→              patchItem(id, { state: "error", error: err.message }, true);
   827→   827→              toast.error(t("downloadFailed"), { description: t(errorKey(err.message)) });
   828→   828→            }
   829→   829→          } finally {
   830→   830→            controllers.current.delete(id);
   831→   831→          }
   832→   832→          return;
   833→   833→        }
   834→   834→        toast.error(t("downloadFailed"), { description: t(errorKey(json.code || "INTERNAL")) });
   835→   835→      } catch {
   836→   836→        toast.error(t("downloadFailed"), { description: "Network error" });
   837→   837→      }
   838→   838→    },
   839→   839→    [patchItem, pollJob, pushHistory, t],
   840→   840→  );
   841→   841→
   842→   842→  return {
   843→   843→    url,
   844→   844→    setUrl,
   845→   845→    phase,
   846→   846→    pendingNote,
   847→   847→    info,
   848→   848→    errorMsg,
   849→   849→    errorCode,
   850→   850→    items,
   851→   851→    history,
   852→   852→    fetchInfo,
   853→   853→    reset,
   854→   854→    startDownload,
   855→   855→    getDirectLink,
   856→   856→    cancelItem,
   857→   857→    removeItem,
   858→   858→    clearFinished,
   859→   859→    clearHistory,
   860→   860→  };
