   106→   106→export function runYtDlp(args: string[], timeoutMs = 120000): Promise<RunResult> {
   107→   107→  return new Promise((resolve, reject) => {
   108→   108→    execFile(
   109→   109→      getYtDlpPath(),
   110→   110→      args,
   111→   111→      { timeout: timeoutMs, maxBuffer: 64 * 1024 * 1024, env: toolEnv() },
   112→   112→      (err, stdout, stderr) => {
   113→   113→        if (err && typeof (err as NodeJS.ErrnoException).code === 'undefined' && !stdout && !stderr) {
   114→   114→          return reject(err)
   115→   115→        }
   116→   116→        resolve({ code: err ? 1 : 0, stdout: stdout ?? '', stderr: stderr ?? '' })
   117→   117→      }
   118→   118→    )
   119→   119→  })
   120→   120→}

   289→   289→interface StoreEntry {

   298→   298→function infoStore(): Map<string, StoreEntry> {
   299→   299→  if (!globalThis.__tgInfoStore) globalThis.__tgInfoStore = new Map()
   300→   300→  return globalThis.__tgInfoStore
   301→   301→}
   302→   302→
   303→   303→const INFO_TTL_MS = 6 * 60 * 60 * 1000
   304→   304→const INFO_MAX = 60
   305→   305→
   306→   306→function pruneStore(): void {
   307→   307→  const store = infoStore()
   308→   308→  const now = Date.now()
   309→   309→  for (const [k, v] of store) {
   310→   310→    if (now - v.at > INFO_TTL_MS) store.delete(k)
   311→   311→  }
   312→   312→  while (store.size > INFO_MAX) {
   313→   313→    const oldest = [...store.entries()].sort((a, b) => a[1].at - b[1].at)[0]
   314→   314→    if (!oldest) break
   315→   315→    store.delete(oldest[0])
   316→   316→  }
   317→   317→}

   319→   319→export function putInfo(raw: RawInfo): string {
   320→   320→  const id = randomUUID()
   321→   321→  const store = infoStore()
   322→   322→  store.set(id, { raw, at: Date.now() })
   323→   323→  pruneStore()
   324→   324→  return id
   325→   325→}

   327→   327→export function getInfo(infoId: string): StoreEntry | null {
   328→   328→  const e = infoStore().get(infoId)
   329→   329→  if (!e) return null
   330→   330→  if (Date.now() - e.at > INFO_TTL_MS) {
   331→   331→    infoStore().delete(infoId)
   332→   332→    return null
   333→   333→  }
   334→   334→  return e
   335→   335→}

   396→   396→  try {
   397→   397→    const { proxiedFetch } = await import('./proxy')

   407→   407→      html.match(/"visitorData":"((?:[^"\\]|\\.)*)"/) ||
   408→   408→      html.match(/"VISITOR_DATA":"((?:[^"\\]|\\.)*)"/)
   409→   409→    if (!m) return null
   410→   410→    const v = JSON.parse(`"${m[1]}"`) as string

   436→   436→export function refreshClientChain(visitorData: string | null): void {
   437→   437→  for (let i = 0; i < CLIENT_ORDER.length; i++) {
   438→   438→    const val = visitorData
   439→   439→      ? `youtube:player_client=${CLIENT_ORDER[i]};visitor_data=${visitorData}`

   441→   441→    CLIENT_CHAIN[i] = ['--extractor-args', val]
   442→   442→  }

   499→   499→async function parseRaw(stdout: string, url: string): Promise<RawInfo> {
   500→   500→  const data = JSON.parse(stdout) as Record<string, unknown>
   501→   501→  const formats = ((data.formats as YtRawFormat[]) ?? []).filter((f) => {
   502→   502→    if (!f.format_id) return false
   503→   503→    if ((f.ext ?? '') === 'mhtml') return false
   504→   504→    if (/storyboard/i.test(f.format_note ?? '')) return false
   505→   505→    return true
   506→   506→  })
   507→   507→  return {
   508→   508→    id: String(data.id ?? extractVideoId(url) ?? ''),
   509→   509→    title: String(data.title ?? 'Untitled'),
   510→   510→    uploader: (data.uploader as string) || (data.channel as string) || undefined,
   511→   511→    channel: (data.channel as string) || undefined,
   512→   512→    duration: (data.duration as number) || undefined,
   513→   513→    view_count: (data.view_count as number) || undefined,
   514→   514→    thumbnail: (data.thumbnail as string) || undefined,
   515→   515→    webpage_url: String(data.webpage_url ?? url),
   516→   516→    is_live: Boolean(data.is_live),
   517→   517→    live_status: (data.live_status as string) || undefined,
   518→   518→    extractor_key: String(data.extractor_key ?? data.extractor ?? 'Generic'),
   519→   519→    formats,
   520→   520→  }
   521→   521→}

   524→   524→ * Full extraction through the current engine stack.
   525→   525→ * Throws DlError with a classified code; extract.ts decides the fallback.
   526→   526→ */
   527→   527→export async function fetchInfo(url: string): Promise<{ infoId: string; raw: RawInfo }> {
   528→   528→  const clean = url.trim()
   529→   529→  if (!/^https?:\/\//i.test(clean)) {
   530→   530→    throw new DlError('INVALID_URL', 'That does not look like a valid video URL.')
   531→   531→  }
   532→   532→
   533→   533→  const { ensurePotServer } = await import('./pot')
   534→   534→  void ensurePotServer().catch(() => undefined)
   535→   535→
   536→   536→  const cookies = await loadCookiesText()
   537→   537→
   538→   538→  if (!isYouTubeUrl(clean)) {
   539→   539→    const { code, stdout, stderr } = await runYtDlp(
   540→   540→      ['-J', ...baseArgs(), ...(cookies ? ['--cookies', cookies] : []), clean],
   541→   541→      150000,
   542→   542→    )
   543→   543→    if (code !== 0 || !stdout.trim().startsWith('{')) {
   544→   544→      const c = classifyStderr(stderr)
   545→   545→      throw new DlError(c.code, c.message, c.hint)
   546→   546→    }
   547→   547→    const raw = await parseRaw(stdout, clean)
   548→   548→    if (raw.is_live) throw new DlError('LIVE', 'Live streams cannot be downloaded.')
   549→   549→    if (raw.formats.length === 0) throw new DlError('NO_FORMATS', 'No downloadable formats found.')
   550→   550→    const infoId = putInfo(raw)
   551→   551→    return { infoId, raw }
   552→   552→  }
   553→   553→
   554→   554→  let visitor = await ensureVisitorData()
   555→   555→  refreshClientChain(visitor)
   556→   556→  let rotatedVisitor = false
   557→   557→  const errors: Classified[] = []
   558→   558→  const MAX_STEPS = 4
   559→   559→
   560→   560→  for (let i = 0; i < Math.min(MAX_STEPS, CLIENT_CHAIN.length); i++) {
   561→   561→    const step = CLIENT_CHAIN[i]
   562→   562→    const { code, stdout, stderr } = await runYtDlp(
   563→   563→      ['-J', ...baseArgs(), ...step, ...(cookies ? ['--cookies', cookies] : []), clean],
   564→   564→      90000,
   565→   565→    )
   566→   566→    if (code === 0 && stdout.trim().startsWith('{')) {
   567→   567→      const raw = await parseRaw(stdout, clean)
   568→   568→      if (raw.is_live) throw new DlError('LIVE', 'Live streams cannot be downloaded.')
   569→   569→      if (raw.formats.length > 0) {
   570→   570→        const infoId = putInfo(raw)
   571→   571→        return { infoId, raw }
   572→   572→      }
   573→   573→      errors.push({ code: 'NO_FORMATS', message: 'Player returned no media formats.' })
   574→   574→      continue
   575→   575→    }
   576→   576→    const c = classifyStderr(stderr || 'empty output')
   577→   577→    if (TERMINAL_CODES.has(c.code)) throw new DlError(c.code, c.message, c.hint)
   578→   578→    errors.push(c)
   579→   579→    // bot wall → rotate visitor identity exactly once and keep rotating clients
   580→   580→    if ((c.code === 'BOT_CHECK' || c.code === 'RATE_LIMITED') && !rotatedVisitor) {
   581→   581→      rotatedVisitor = true
   582→   582→      const minted = await mintVisitorData()
   583→   583→      if (minted && minted !== visitor) {
   584→   584→        visitor = minted
   585→   585→        persistVisitorData(minted)
   586→   586→        refreshClientChain(visitor)
   587→   587→      }
   588→   588→    }
   589→   589→  }
   590→   590→
   591→   591→  const worst = errors.find((e) => e.code === 'BOT_CHECK') ?? errors[0] ?? { code: 'FETCH_FAILED' as const, message: 'Extraction failed.' }
   592→   592→  throw new DlError(worst.code, worst.message, worst.hint)
   593→   593→}

   606→   606→export async function resolveFormatUrl(webpageUrl: string, selector: string): Promise<string | null> {
   607→   607→  try {
   608→   608→    const cookies = await loadCookiesText()
   609→   609→    const { code, stdout } = await runYtDlp(
   610→   610→      [
   611→   611→        ...baseArgs(),
   612→   612→        ...(cookies ? ['--cookies', cookies] : []),
   613→   613→        ...CLIENT_CHAIN[0],
   614→   614→        '-f',
   615→   615→        selector,
   616→   616→        '--get-url',
   617→   617→        webpageUrl,
   618→   618→      ],
   619→   619→      90000,
   620→   620→    )
   621→   621→    if (code !== 0) return null
   622→   622→    const line = stdout
   623→   623→      .split('\n')
   624→   624→      .map((l) => l.trim())
   625→   625→      .find((l) => /^https?:\/\//.test(l))
   626→   626→    return line ?? null
   627→   627→  } catch {
   628→   628→    return null
   629→   629→  }

   634→   634→function codecLabel(c?: string): string | undefined {
   635→   635→  if (!c || c === 'none') return undefined
   636→   636→  const s = c.toLowerCase()
   637→   637→  if (s.startsWith('avc1') || s.startsWith('avc3') || s.includes('h264') || s.includes('h.264')) return 'H.264'
   638→   638→  if (s.startsWith('vp9') || s.includes('vp09')) return 'VP9'
   639→   639→  if (s.startsWith('vp8')) return 'VP8'
   640→   640→  if (s.startsWith('av01')) return 'AV1'
   641→   641→  if (s.startsWith('mp4a') || s.includes('aac')) return 'AAC'
   642→   642→  if (s.includes('opus')) return 'Opus'
   643→   643→  if (s.includes('vorbis')) return 'Vorbis'
   644→   644→  if (s.includes('ec-3') || s.includes('eac3')) return 'E-AC3'
   645→   645→  if (s.includes('ac-3')) return 'AC3'
   646→   646→  return c.split('.')[0].toUpperCase()
   647→   647→}

   653→   653→function videoQuality(f: YtRawFormat): string {
   654→   654→  const h = f.height
   655→   655→  if (!h) return f.format_note || f.format_id
   656→   656→  const fps = f.fps && f.fps > 30 ? f.fps : 0
   657→   657→  return `${h}p${fps || ''}`

   660→   660→function audioQuality(f: YtRawFormat): string {
   661→   661→  const k = Math.round(f.abr || f.tbr || 0)
   662→   662→  return k > 0 ? `${k}k` : f.format_note || f.format_id
   663→   663→}

   665→   665→function isStreamable(f: YtRawFormat): boolean {
   666→   666→  const proto = (f.protocol ?? '').toLowerCase()
   667→   667→  return proto === 'https' || proto === 'http' || proto === ''
   668→   668→}

   670→   670→function pickAudio(formats: YtRawFormat[], container: string): YtRawFormat | null {
   671→   671→  const audios = formats.filter((f) => f.vcodec === 'none' && f.acodec && f.acodec !== 'none')
   672→   672→  if (audios.length === 0) return null
   673→   673→  const family = /webm|mkv/.test(container) ? ['opus', 'vorbis'] : ['mp4a', 'aac']
   674→   674→  const scored = audios
   675→   675→    .map((f) => {
   676→   676→      const s = (f.acodec ?? '').toLowerCase()
   677→   677→      const fam = family.some((x) => s.includes(x)) ? 1 : 0
   678→   678→      return { f, score: fam * 1000 + (f.abr ?? f.tbr ?? 0) }
   679→   679→    })
   680→   680→    .sort((a, b) => b.score - a.score)
   681→   681→  return scored[0].f
   682→   682→}

   684→   684→export function toPublicInfo(infoId: string, raw: RawInfo): PublicInfo {
   685→   685→  const isYT = /youtube/i.test(raw.extractor_key ?? '') || isYouTubeUrl(raw.webpage_url)
   686→   686→  const source = /:piped/i.test(raw.extractor_key ?? '')
   687→   687→    ? 'Piped'
   688→   688→    : /:invidious/i.test(raw.extractor_key ?? '')
   689→   689→      ? 'Invidious'
   690→   690→      : isYT
   691→   691→        ? 'youtube'
   692→   692→        : 'other'
   693→   693→
   694→   694→  const thumbProxy = (u?: string): string => {
   695→   695→    if (!u || !/^https?:\/\//.test(u)) return ''
   696→   696→    return `/api/thumb?url=${encodeURIComponent(u)}`
   697→   697→  }
   698→   698→
   699→   699→  const instant: PublicFormat[] = []
   700→   700→  const videoOnly: Array<{ f: YtRawFormat; fmt: PublicFormat }> = []
   701→   701→  const audio: PublicFormat[] = []
   702→   702→
   703→   703→  for (const f of raw.formats) {
   704→   704→    const hasV = f.vcodec && f.vcodec !== 'none'
   705→   705→    const hasA = f.acodec && f.acodec !== 'none'
   706→   706→    const ext = (f.ext ?? 'mp4').toLowerCase()
   707→   707→    const size = fmtSize(f)
   708→   708→
   709→   709→    if (hasV && hasA) {
   710→   710→      // muxed — instant single-stream
   711→   711→      if (!isStreamable(f)) continue
   712→   712→      instant.push({
   713→   713→        id: f.format_id,
   714→   714→        kind: 'video',
   715→   715→        quality: videoQuality(f),
   716→   716→        container: ext === 'webm' ? 'webm' : 'mp4',
   717→   717→        vcodec: codecLabel(f.vcodec),
   718→   718→        acodec: codecLabel(f.acodec),
   719→   719→        fps: f.fps,
   720→   720→        filesize: size,
   721→   721→        approx: !f.filesize,
   722→   722→        hasAudio: true,
   723→   723→        instant: true,
   724→   724→        needsMerge: false,
   725→   725→        hdr: /hdr/i.test(f.dynamic_range ?? ''),
   726→   726→        recommended: true,
   727→   727→      })
   728→   728→    } else if (hasV) {
   729→   729→      const container = ext === 'webm' ? 'webm' : 'mp4'
   730→   730→      const fmt: PublicFormat = {
   731→   731→        id: f.format_id, // replaced by composite below
   732→   732→        kind: 'video',
   733→   733→        quality: videoQuality(f),
   734→   734→        container,
   735→   735→        vcodec: codecLabel(f.vcodec),
   736→   736→        fps: f.fps,
   737→   737→        filesize: size,
   738→   738→        approx: !f.filesize,
   739→   739→        hasAudio: false,
   740→   740→        instant: false,
   741→   741→        needsMerge: true,
   742→   742→        hdr: /hdr/i.test(f.dynamic_range ?? ''),
   743→   743→      }
   744→   744→      videoOnly.push({ f, fmt })
   745→   745→    } else if (hasA) {
   746→   746→      if (!isStreamable(f)) continue
   747→   747→      audio.push({
   748→   748→        id: f.format_id,
   749→   749→        kind: 'audio',
   750→   750→        quality: audioQuality(f),
   751→   751→        container: ext === 'webm' ? 'webm' : ext === 'm4a' ? 'm4a' : ext,
   752→   752→        acodec: codecLabel(f.acodec),
   753→   753→        filesize: size,
   754→   754→        approx: !f.filesize,
   755→   755→        hasAudio: true,
   756→   756→        instant: true,
   757→   757→        needsMerge: false,
   758→   758→      })
   759→   759→    }
   760→   760→  }
   761→   761→
   762→   762→  // composite merges: pair each video-only track with its best audio partner
   763→   763→  const composites: PublicFormat[] = videoOnly.map(({ f, fmt }) => {
   764→   764→    const a = pickAudio(raw.formats, fmt.container)
   765→   765→    const id = a ? `${f.format_id}|${a.format_id}|${fmt.container}` : f.format_id
   766→   766→    const total = (fmt.filesize ?? 0) + (a ? fmtSize(a) ?? 0 : 0)
   767→   767→    return {
   768→   768→      ...fmt,
   769→   769→      id,
   770→   770→      filesize: total > 0 ? total : fmt.filesize,
   771→   771→      approx: !f.filesize || !a?.filesize,
   772→   772→    }
   773→   773→  })
   774→   774→
   775→   775→  const rankVideo = (f: PublicFormat): number => {
   776→   776→    const h = parseInt(f.quality, 10) || 0
   777→   777→    return h * 100 + (f.fps ?? 0) * 2 + (f.container === 'mp4' ? 1 : 0)
   778→   778→  }
   779→   779→  instant.sort((a, b) => rankVideo(b) - rankVideo(a))
   780→   780→  composites.sort((a, b) => rankVideo(b) - rankVideo(a))
   781→   781→  audio.sort((a, b) => (parseInt(b.quality, 10) || 0) - (parseInt(a.quality, 10) || 0))
   782→   782→
   783→   783→  return {
   784→   784→    infoId,
   785→   785→    title: raw.title,
   786→   786→    author: raw.uploader || raw.channel || '',
   787→   787→    views: raw.view_count ?? 0,
   788→   788→    duration: raw.duration ?? 0,
   789→   789→    thumbnail: thumbProxy(raw.thumbnail),
   790→   790→    webpageUrl: raw.webpage_url,
   791→   791→    source,
   792→   792→    videoFormats: [...composites, ...instant],
   793→   793→    audioFormats: audio,
   794→   794→    mp3Options: hasFfmpeg() ? [320, 192, 128] : [],
   795→   795→  }
   796→   796→}
