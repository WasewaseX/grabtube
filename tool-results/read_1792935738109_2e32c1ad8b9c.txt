     1→     1→/**
     2→     2→ * Server-side download jobs — the MIDDLEMAN engine.
     3→     3→ *
     4→     4→ * The server (fast datacenter pipe + WARP egress) always downloads from
     5→     5→ * YouTube FIRST, at full speed. The user's browser/downloader then pulls the
     6→     6→ * finished file from this server with Range/resume support — YouTube is never
     7→     7→ * involved in the user's own connection (crucial for slow/censored networks).
     8→     8→ *
     9→     9→ *  • merge / mp3    – yt-dlp downloads the selected tracks (optionally merging
    10→    10→ *                     with ffmpeg) and reports --newline progress.
    11→    11→ *  • file           – yt-dlp downloads ONE format (muxed video or native
    12→    12→ *                     audio) server-side. Used for "instant" formats so the
    13→    13→ *                     user never live-streams from YouTube through the app.
    14→    14→ *  • urlmerge       – ffmpeg pulls the video+audio direct URLs itself and
    15→    15→ *                     stream-copies them into one container. Used when the
    16→    16→ *                     format list came from a proxy provider (Piped/Invidious).
    17→    17→ *  • urlaudio       – ffmpeg pulls an audio direct URL and converts to MP3.
    18→    18→ *  • urlfile        – ffmpeg pulls a single direct URL and remuxes to the
    19→    19→ *                     target container (proxy-source instant formats).
    20→    20→ *
    21→    21→ * Jobs write to a temp file while emitting live progress, then the client
    22→    22→ * streams the finished file from /api/jobs/[id]/file. Finished files are kept
    23→    23→ * for hours (touch-extended while actively being downloaded) so slow
    24→    24→ * connections can pause/resume and external managers can re-hit the link.
    25→    25→ */

   130→   130→export function ensureCleanupTimer(): void {
   131→   131→  if (globalThis.__tgJobsCleanup) return;
   132→   132→  globalThis.__tgJobsCleanup = setInterval(() => {
   133→   133→    const store = jobStore();
   134→   134→    const now = Date.now();
   135→   135→    for (const [id, job] of store) {
   136→   136→      const expired = now - job.createdAt > JOBS_TTL_MS;
   137→   137→      const dead =
   138→   138→        (job.state === "cancelled" || job.state === "error") && now - job.createdAt > 10 * 60 * 1000;
   139→   139→      // actively downloading server-side → never touch (a 2160p merge can
   140→   140→      // legitimately run past an hour; killing it corrupts the transfer)
   141→   141→      const running =
   142→   142→        (job.state === "downloading" || job.state === "merging") &&
   143→   143→        now - job.createdAt < 3 * 60 * 60 * 1000;
   144→   144→      // an external manager is mid-download of the finished file → keep it
   145→   145→      const servedRecently = job.lastServeAt != null && now - job.lastServeAt < ACTIVE_SERVE_MS;
   146→   146→      if ((expired && !running && !servedRecently) || dead) {
   147→   147→        if (job.proc) {
   148→   148→          try {
   149→   149→            job.proc.kill("SIGKILL");
   150→   150→          } catch {
   151→   151→            /* noop */
   152→   152→          }
   153→   153→        }
   154→   154→        // registered files (direct links) are owned by the store's eviction —
   155→   155→        // deleting them here would break links that are still being used
   156→   156→        if (job.filePath && existsSync(job.filePath) && !job.dlKey) {
   157→   157→          try {
   158→   158→            rmSync(job.filePath, { force: true });
   159→   159→          } catch {
   160→   160→            /* noop */
   161→   161→          }
   162→   162→        }
   163→   163→        store.delete(id);
   164→   164→      }
   165→   165→    }
   166→   166→  }, 60_000);
   167→   167→  globalThis.__tgJobsCleanup.unref();
   168→   168→}
   169→   169→
   170→   170→/* ---------------- stall watchdog ---------------- */
   171→   171→
   172→   172→/**
   173→   173→ * WEDGE KILLER — the one failure mode yt-dlp cannot self-heal: through a
   174→   174→ * lossy tunnel a fragment socket occasionally freezes in a read that never
   175→   175→ * trips the socket timeout (data trickles byte-wise). The process then sits
   176→   176→ * "downloading" forever with a frozen .part file. Real download managers
   177→   177→ * solve this with a stall watchdog: no byte progress for N minutes → kill
   178→   178→ * the process → the close handler re-queues the job (transient) and the
   179→   179→ * retry RESUMES from the .part file. Bytes already fetched are never lost.
   180→   180→ */
   181→   181→const STALL_KILL_MS = 4 * 60 * 1000; // no byte movement for 4 min → wedge
   182→   182→const HARD_CAP_MS = 60 * 60 * 1000; // 1h without finishing → re-queue
   183→   183→
   184→   184→function ensureStallWatchdog(): void {
   185→   185→  if (globalThis.__tgJobsWatchdog) return;
   186→   186→  globalThis.__tgJobsWatchdog = setInterval(() => {
   187→   187→    const now = Date.now();
   188→   188→    for (const job of jobStore().values()) {
   189→   189→      if (job.state !== "downloading" && job.state !== "merging") continue;
   190→   190→      const started = job.lastProgressAt ?? job.createdAt;
   191→   191→      const stalled = now - started > STALL_KILL_MS && (job.progress > 0 || job.receivedBytes > 0);
   192→   192→      const overCap = now - started > HARD_CAP_MS;
   193→   193→      if (!stalled && !overCap) continue;
   194→   194→      console.log(
   195→   195→        `[jobs] job ${job.id} watchdog: ${stalled ? "stalled" : "over cap"} — killing for resume-retry`,
   196→   196→      );
   197→   197→      job.lastProgressAt = now; // avoid double-fire while the close event lands
   198→   198→      try {
   199→   199→        job.proc?.kill("SIGKILL");
   200→   200→      } catch {
   201→   201→        /* noop */
   202→   202→      }
   203→   203→    }
   204→   204→  }, 30_000);
   205→   205→  globalThis.__tgJobsWatchdog.unref();
   206→   206→}

   212→   212→/* ---------------- concurrency pump ---------------- */
   213→   213→function activeCount(): number {
   214→   214→  let active = 0;
   215→   215→  for (const job of jobStore().values()) {
   216→   216→    if (job.state === "downloading" || job.state === "merging") active++;
   217→   217→  }
   218→   218→  return active;
   219→   219→}
   220→   220→
   221→   221→function pump(): void {
   222→   222→  const store = jobStore();
   223→   223→  let active = activeCount();
   224→   224→  for (const job of store.values()) {
   225→   225→    if (active >= MAX_CONCURRENT_JOBS) break;
   226→   226→    if (job.state === "queued" && job.args) {
   227→   227→      const args = job.args;
   228→   228→      job.args = undefined;
   229→   229→      active++;
   230→   230→      if (job.engine === "ffmpeg") spawnFfmpegJob(job, args);
   231→   231→      else spawnYtdlpJob(job, args);
   232→   232→    }
   233→   233→  }
   234→   234→}

   236→   236→function spawnYtdlpJob(job: Job, args: string[]): void {
   237→   237→  const engine = getEngine();
   238→   238→  job.lastArgs = args;
   239→   239→  job.lastProgressAt = Date.now();
   240→   240→  const proc = spawn(engine.cmd, [...engine.baseArgs, ...args], {
   241→   241→    stdio: ["ignore", "pipe", "pipe"],
   242→   242→    // deno (JS-challenge solver) + venv binaries must be on PATH or YouTube
   243→   243→    // answers with an image-only format list
   244→   244→    env: toolEnv(),
   245→   245→  });
   246→   246→  job.proc = proc;
   247→   247→  job.state = "downloading";
   248→   248→  job.phase = "download";
   249→   249→  void attachYtdlpOutput(job, proc);
   250→   250→}

   252→   252→function spawnFfmpegJob(job: Job, args: string[]): void {
   253→   253→  job.lastArgs = args;
   254→   254→  job.lastProgressAt = Date.now();
   255→   255→  const proc = spawn("ffmpeg", ["-hide_banner", "-loglevel", "error", ...args], {

   262→   262→  job.state = "downloading";
   263→   263→  job.phase = "download";
   264→   264→  void attachFfmpegOutput(job, proc);
   265→   265→}

   314→   314→export function createJob(input: CreateJobInput): Job {
   315→   315→  const entry = getInfo(input.infoId);
   316→   316→  if (!entry) throw new DlError("EXPIRED", "This video's session expired. Fetch it again.");
   317→   317→
   318→   318→  // disk pre-flight: merge jobs buffer video+audio in tmp before muxing — a
   319→   319→  // full disk otherwise surfaces as a cryptic ENOSPC error at 98% done
   320→   320→  try {
   321→   321→    const s = statfsSync(jobsDir());
   322→   322→    const freeBytes = s.bsize * s.bavail;
   323→   323→    if (freeBytes < 1.2 * 1024 * 1024 * 1024) {
   324→   324→      throw new DlError(
   325→   325→        "DISK_FULL",
   326→   326→        "Server disk is almost full — clear finished downloads (or old job files) and try again.",
   327→   327→      );
   328→   328→    }
   329→   329→  } catch (e) {
   330→   330→    if (e instanceof DlError) throw e;
   331→   331→    /* statfs unavailable — skip the guard */
   332→   332→  }
   333→   333→
   334→   334→  const raw = entry.raw;
   335→   335→  const title = sanitizeFilename(input.title || raw.title || "video");
   336→   336→  const durationSec = input.duration || raw.duration || 0;
   337→   337→
   338→   338→  const jobId = randomUUID();
   339→   339→  let ext: string;
   340→   340→  let args: string[];
   341→   341→  let engine: Job["engine"];
   342→   342→  let mode = input.mode;
   343→   343→
   344→   344→  if (mode === "mp3") {
   345→   345→    // mode "mp3" covers every audio conversion target (mp3/m4a/opus/flac/wav)
   346→   346→    const target = ["mp3", "m4a", "opus", "flac", "wav"].includes(input.audioFormat ?? "")
   347→   347→      ? (input.audioFormat as string)
   348→   348→      : "mp3";
   349→   349→    ext = target;
   350→   350→    engine = "ytdlp";
   351→   351→    const bitrate = [320, 192, 128].includes(input.bitrate || 0) ? input.bitrate : 320;
   352→   352→    args = [
   353→   353→      ...baseArgs(),
   354→   354→      // same client chain as extraction, or the job re-resolves on a different
   355→   355→      // player client and the selection breaks (format roulette)
   356→   356→      ...CLIENT_CHAIN[0],
   357→   357→      "--no-playlist",
   358→   358→      "--newline",
   359→   359→      "-N",
   360→   360→      "8",
   361→   361→      "-f",
   362→   362→      "bestaudio/best",
   363→   363→      "-x",
   364→   364→      "--audio-format",
   365→   365→      target,
   366→   366→      // lossless/pcm targets ignore quality; mp3 uses the chosen bitrate,
   367→   367→      // everything else re-encodes at the best sensible preset (0)
   368→   368→      "--audio-quality",
   369→   369→      target === "mp3" ? `${bitrate}K` : "0",
   370→   370→      "-o",
   371→   371→      path.join(jobsDir(), `${jobId}.%(ext)s`),
   372→   372→      input.webpageUrl || raw.webpage_url,
   373→   373→    ];
   374→   374→  } else if (mode === "file") {
   375→   375→    // single-format server-side download (muxed video / native audio) —
   376→   376→    // "instant" formats go through this too so the user NEVER live-streams
   377→   377→    // from YouTube through the app; they always pull a local, resumable file
   378→   378→    const fmtId = (input.videoFormatId || "").split("|")[0];
   379→   379→    if (!fmtId) throw new DlError("INVALID_URL", "Missing format id.");
   380→   380→    const stored = raw.formats.find((f) => f.format_id === fmtId);
   381→   381→    ext = (stored?.ext || "mp4").toLowerCase();
   382→   382→    engine = "ytdlp";
   383→   383→    args = [
   384→   384→      ...baseArgs(),
   385→   385→      ...CLIENT_CHAIN[0],
   386→   386→      "--no-playlist",
   387→   387→      "--newline",
   388→   388→      "-N",
   389→   389→      "8",
   390→   390→      "-f",
   391→   391→      `${fmtId}/best`,
   392→   392→      "-o",
   393→   393→      path.join(jobsDir(), `${jobId}.%(ext)s`),
   394→   394→      input.webpageUrl || raw.webpage_url,
   395→   395→    ];
   396→   396→  } else if (mode === "urlfile") {
   397→   397→    // proxy-provider single stream → ffmpeg remux into the target container
   398→   398→    if (!input.urls?.video) throw new DlError("INVALID_URL", "Missing stream URL.");
   399→   399→    const fmtId = (input.videoFormatId || "").split("|")[0];
   400→   400→    const stored = raw.formats.find((f) => f.format_id === fmtId);
   401→   401→    ext = (stored?.ext || "mp4").toLowerCase();
   402→   402→    engine = "ffmpeg";
   403→   403→    const out = path.join(jobsDir(), `${randomUUID()}.${ext}`);
   404→   404→    const isMp4 = ext === "mp4";
   405→   405→    args = [
   406→   406→      "-y",
   407→   407→      "-i",
   408→   408→      input.urls.video,
   409→   409→      "-c",
   410→   410→      "copy",
   411→   411→      ...(isMp4 ? ["-movflags", "+faststart"] : []),
   412→   412→      "-progress",
   413→   413→      "pipe:1",
   414→   414→      out,
   415→   415→    ];
   416→   416→  } else if (mode === "urlaudio") {
   417→   417→    if (!input.urls?.audio) throw new DlError("INVALID_URL", "Missing audio stream URL.");
   418→   418→    ext = "mp3";
   419→   419→    engine = "ffmpeg";
   420→   420→    const bitrate = [320, 192, 128].includes(input.bitrate || 0) ? input.bitrate : 192;
   421→   421→    const out = path.join(jobsDir(), `${randomUUID()}.mp3`);
   422→   422→    args = ["-y", "-i", input.urls.audio, "-vn", "-c:a", "libmp3lame", "-b:a", `${bitrate}K`, "-progress", "pipe:1", out];
   423→   423→  } else if (mode === "urlmerge") {
   424→   424→    if (!input.urls?.video || !input.urls?.audio)
   425→   425→      throw new DlError("INVALID_URL", "Missing stream URLs for merge.");
   426→   426→    const [vId, aId, container] = (input.videoFormatId || "").split("|");
   427→   427→    ext = ["mp4", "webm", "mkv"].includes(container) ? container : "mkv";
   428→   428→    engine = "ffmpeg";
   429→   429→    const out = path.join(jobsDir(), `${randomUUID()}.${ext}`);
   430→   430→    const isMp4 = ext === "mp4";
   431→   431→    args = [
   432→   432→      "-y",
   433→   433→      "-i",
   434→   434→      input.urls.video,
   435→   435→      "-i",
   436→   436→      input.urls.audio,
   437→   437→      "-c",
   438→   438→      "copy",
   439→   439→      ...(isMp4 ? ["-movflags", "+faststart"] : []),
   440→   440→      "-progress",
   441→   441→      "pipe:1",
   442→   442→      out,
   443→   443→    ];
   444→   444→    void vId;
   445→   445→    void aId;
   446→   446→  } else {
   447→   447→    /* mode === "merge" (yt-dlp) */
   448→   448→    const [vId, aId, container] = (input.videoFormatId || "").split("|");
   449→   449→    if (!vId || !aId || !container) throw new DlError("INVALID_URL", "Bad merge request.");
   450→   450→    ext = ["mp4", "webm", "mkv"].includes(container) ? container : "mkv";
   451→   451→    engine = "ytdlp";
   452→   452→    const tmpPath = path.join(jobsDir(), `${jobId}.%(ext)s`);
   453→   453→    // resilient selector: the exact itags first, then quality-matched
   454→   454→    // fallbacks — yt-dlp's client roulette means the same itag may not exist
   455→   455→    // on the client that answers the job's own re-resolve
   456→   456→    const vf = raw.formats.find((f) => f.format_id === vId);
   457→   457→    const h = vf?.height;
   458→   458→    const fps = vf?.fps;
   459→   459→    const qualityFallback = h
   460→   460→      ? `bestvideo[height<=${h}${fps ? `][fps<=${fps}` : ""}]+bestaudio/bestvideo[height<=${h}]+bestaudio`
   461→   461→      : "bestvideo+bestaudio";
   462→   462→    args = [
   463→   463→      ...baseArgs(),
   464→   464→      ...CLIENT_CHAIN[0],
   465→   465→      "--no-playlist",
   466→   466→      "--newline",
   467→   467→      "-N",
   468→   468→      "8",
   469→   469→      "-f",
   470→   470→      `${vId}+${aId}/${qualityFallback}/bestvideo+bestaudio/best`,
   471→   471→      "--merge-output-format",
   472→   472→      ext,
   473→   473→      "-o",
   474→   474→      tmpPath,
   475→   475→      input.webpageUrl || raw.webpage_url,
   476→   476→    ];
   477→   477→  }
   478→   478→
   479→   479→  const job: Job = {
   480→   480→    id: jobId,
   481→   481→    title,
   482→   482→    filename: `${title}.${ext}`,
   483→   483→    ext,
   484→   484→    mime: mimeFor(ext),
   485→   485→    state: "queued",
   486→   486→    phase: "queue",
   487→   487→    progress: 0,
   488→   488→    speed: null,
   489→   489→    eta: null,
   490→   490→    receivedBytes: 0,
   491→   491→    sizeBytes: null,
   492→   492→    error: null,
   493→   493→    mode,
   494→   494→    createdAt: Date.now(),
   495→   495→    filePath: null,
   496→   496→    proc: null,
   497→   497→    lastServeAt: null,
   498→   498→    engine,
   499→   499→    durationSec,
   500→   500→  };
   501→   501→
   502→   502→  const fullJob: Job = {
   503→   503→    ...job,
   504→   504→    dlKey: input.dlKey,
   505→   505→    webpageUrl: input.webpageUrl || raw.webpage_url,
   506→   506→    qualityLabel: input.qualityLabel,
   507→   507→    formatSpec: input.videoFormatId || input.audioFormat,
   508→   508→  };
   509→   509→
   510→   510→  jobStore().set(fullJob.id, fullJob);
   511→   511→  ensureCleanupTimer();
   512→   512→  ensureStallWatchdog();
   513→   513→  fullJob.args = args;
   514→   514→  pump();
   515→   515→
   516→   516→  return fullJob;
   517→   517→}

   519→   519→/**
   520→   520→ * Direct-link jobs: persist the finished file into the StoredFile registry
   521→   521→ * (async, non-blocking). Once registered, /api/file/<token> serves the LOCAL
   522→   522→ * file forever (until eviction) — that token is pinned onto the job so poll
   523→   523→ * clients pick it up.
   524→   524→ *
   525→   525→ * The finished file is FIRST moved out of /tmp (tmp cleaners would break
   526→   526→ * every stored link at once) into the durable downloads/ directory — the
   527→   527→ * "database" the direct links point at. Only then is it registered.
   528→   528→ */
   529→   529→function persistentDownloadsPath(filename: string): string {
   530→   530→  const dir = "/home/z/my-project/downloads";
   531→   531→  try {
   532→   532→    if (!existsSync(dir)) mkdirSync(dir, { recursive: true });
   533→   533→  } catch {
   534→   534→    return path.join(jobsDir(), filename); // fall back to the tmp location
   535→   535→  }
   536→   536→  let target = path.join(dir, filename);
   537→   537→  const ext = path.extname(filename);
   538→   538→  const base = filename.slice(0, filename.length - ext.length);
   539→   539→  for (let i = 2; existsSync(target); i++) {
   540→   540→    target = path.join(dir, `${base} (${i})${ext}`);
   541→   541→  }
   542→   542→  return target;
   543→   543→}
   544→   544→
   545→   545→function moveIntoDownloads(job: Job): void {
   546→   546→  if (!job.filePath || !existsSync(job.filePath)) return;
   547→   547→  const target = persistentDownloadsPath(job.filename);
   548→   548→  if (target === job.filePath) return;
   549→   549→  try {
   550→   550→    try {
   551→   551→      renameSync(job.filePath, target); // fast path: same filesystem
   552→   552→    } catch {
   553→   553→      // cross-device (different mount) → copy + delete
   554→   554→      copyFileSync(job.filePath, target);
   555→   555→      rmSync(job.filePath, { force: true });
   556→   556→    }
   557→   557→    job.filePath = target;
   558→   558→  } catch {
   559→   559→    /* move failed — register the tmp file anyway; self-heal re-fetches */
   560→   560→  }
   561→   561→}
   562→   562→
   563→   563→function registerFinishedFile(job: Job): void {
   564→   564→  if (!job.dlKey || !job.filePath) return;
   565→   565→  moveIntoDownloads(job);
   566→   566→  void registerStoredFile({
   567→   567→    lookupKey: job.dlKey,
   568→   568→    filePath: job.filePath,
   569→   569→    filename: job.filename,
   570→   570→    mime: job.mime,
   571→   571→    kind: job.mode === "mp3" ? "audio" : "video",
   572→   572→    quality: job.qualityLabel,
   573→   573→    title: job.title,
   574→   574→    webpageUrl: job.webpageUrl,
   575→   575→    formatSpec: job.formatSpec,
   576→   576→  }).then((info) => {
   577→   577→    if (info) {
   578→   578→      job.directToken = info.token;
   579→   579→      console.log(`[jobs] file registered → /api/file/${info.token} (${job.filename})`);
   580→   580→    }
   581→   581→  });
   582→   582→}

   584→   584→/* ---------------- self-healing retry ---------------- */
   585→   585→
   586→   586→/**
   587→   587→ * Re-queue a job that died from a TRANSIENT error. The egress lane is
   588→   588→ * re-probed first (a recovered tunnel gets picked for the retry) and yt-dlp
   589→   589→ * resumes the .part file it already wrote — no bytes re-downloaded.
   590→   590→ * Terminal errors (private video, age gate, bad URL…) never retry.
   591→   591→ */
   592→   592→function scheduleJobRetry(job: Job, errCode: string): boolean {
   593→   593→  if (TERMINAL_JOB_CODES.has(errCode)) return false;
   594→   594→  if ((job.retries ?? 0) >= MAX_JOB_RETRIES) return false;
   595→   595→  if (!job.lastArgs || job.lastArgs.length === 0) return false;
   596→   596→  job.retries = (job.retries ?? 0) + 1;
   597→   597→  job.state = "queued";
   598→   598→  job.phase = "queue";
   599→   599→  job.progress = 0;
   600→   600→  job.error = null;
   601→   601→  job.speed = null;
   602→   602→  job.eta = null;
   603→   603→  job.createdAt = Date.now(); // reset the TTL clock for the new attempt
   604→   604→  console.log(
   605→   605→    `[jobs] job ${job.id} transient failure (${errCode}) — retry ${job.retries}/${MAX_JOB_RETRIES}`,
   606→   606→  );
   607→   607→  const delay = 4_000 * job.retries;
   608→   608→  setTimeout(() => {
   609→   609→    // re-probe lanes so a recovered tunnel is active before the retry spawns
   610→   610→    void probeAll().finally(() => {
   611→   611→      job.args = job.lastArgs;
   612→   612→      pump();
   613→   613→    });
   614→   614→  }, delay);
   615→   615→  return true;
   616→   616→}

   619→   619→
   620→   620→async function attachYtdlpOutput(job: Job, proc: ChildProcess): Promise<void> {
   621→   621→  let stderr = "";
   622→   622→
   623→   623→  proc.stdout?.on("data", (chunk: Buffer) => {
   624→   624→    for (const line of chunk.toString().split("\n")) {
   625→   625→      const pct = line.match(/\[download\]\s+([\d.]+)%/);
   626→   626→      if (pct) {
   627→   627→        job.state = "downloading";
   628→   628→        job.phase = "download";
   629→   629→        job.lastProgressAt = Date.now();
   630→   630→        job.progress = Math.min(0.97, parseFloat(pct[1]) / 100);
   631→   631→        const spd = line.match(/at\s+([\d.]+\w*\/s)/);
   632→   632→        if (spd) job.speed = spd[1];
   633→   633→        const eta = line.match(/ETA\s+([\d:]+)/);
   634→   634→        if (eta) job.eta = eta[1];
   635→   635→      }
   636→   636→      if (line.includes("[Merger]") || line.includes("[ExtractAudio]")) {
   637→   637→        job.state = "merging";
   638→   638→        job.phase = "merge";
   639→   639→        job.progress = 0.985;
   640→   640→      }
   641→   641→    }
   642→   642→  });
   643→   643→
   644→   644→  proc.stderr?.on("data", (chunk: Buffer) => {
   645→   645→    stderr += chunk.toString();
   646→   646→    if (stderr.length > 32_000) stderr = stderr.slice(-16_000);
   647→   647→  });
   648→   648→
   649→   649→  proc.on("error", () => {
   650→   650→    if (job.state === "cancelled") return;
   651→   651→    job.state = "error";
   652→   652→    job.error = "INTERNAL";
   653→   653→    pump();
   654→   654→  });
   655→   655→
   656→   656→  proc.on("close", (code) => {
   657→   657→    if (job.state === "cancelled") return;
   658→   658→    if (code === 0) {
   659→   659→      const candidates = [job.ext, "mp4", "webm", "mkv", "m4a", "mp3"].map((e) =>
   660→   660→        path.join(jobsDir(), `${job.id}.${e}`),
   661→   661→      );
   662→   662→      // yt-dlp output templates used a random uuid (not job.id) → scan by ext
   663→   663→      const found = candidates.find((c) => existsSync(c));
   664→   664→      if (found) {
   665→   665→        job.filePath = found;
   666→   666→        job.sizeBytes = statSync(found).size;
   667→   667→        job.state = "ready";
   668→   668→        job.phase = "done";
   669→   669→        job.progress = 1;
   670→   670→        job.speed = null;
   671→   671→        job.eta = null;
   672→   672→        registerFinishedFile(job);
   673→   673→      } else {
   674→   674→        job.state = "error";
   675→   675→        job.error = "INTERNAL";
   676→   676→      }
   677→   677→    } else {
   678→   678→      const err = classifyStderr(stderr);
   679→   679→      if (scheduleJobRetry(job, err.code)) {
   680→   680→        pump();
   681→   681→        return;
   682→   682→      }
   683→   683→      job.state = "error";
   684→   684→      job.error = err.code;
   685→   685→      console.error(`[jobs] yt-dlp job ${job.id} failed (${err.code}): ${stderr.slice(-400).replace(/\n/g, " | ")}`);
   686→   686→    }
   687→   687→    pump();
   688→   688→  });
   689→   689→}

   691→   691→async function attachFfmpegOutput(job: Job, proc: ChildProcess): Promise<void> {
   692→   692→  let stderr = "";
   693→   693→
   694→   694→  proc.stdout?.on("data", (chunk: Buffer) => {
   695→   695→    // ffmpeg -progress pipe:1 emits key=value lines
   696→   696→    for (const line of chunk.toString().split("\n")) {
   697→   697→      const m = line.match(/out_time_us=(\d+)/) || line.match(/out_time_ms=(\d+)/);
   698→   698→      if (m && job.durationSec > 0) {
   699→   699→        const done = parseInt(m[1], 10) / 1_000_000;
   700→   700→        job.lastProgressAt = Date.now();
   701→   701→        job.progress = Math.min(0.98, done / job.durationSec);
   702→   702→        job.state = job.progress >= 0.98 ? "merging" : "downloading";
   703→   703→        job.phase = job.progress >= 0.98 ? "merge" : "download";
   704→   704→      }
   705→   705→      const spd = line.match(/speed=\s*([\d.]+)x/);
   706→   706→      if (spd) {
   707→   707→        const x = parseFloat(spd[1]);
   708→   708→        if (x > 0) job.speed = `${x.toFixed(1)}×`;
   709→   709→      }
   710→   710→      const sz = line.match(/total_size=(\d+)/);
   711→   711→      if (sz) job.receivedBytes = parseInt(sz[1], 10);
   712→   712→      const rem = line.match(/out_time=(\S+)/);
   713→   713→      if (rem && job.durationSec > 0) {
   714→   714→        const parts = rem[1].split(":").map(parseFloat);
   715→   715→        const done = parts.length === 3 ? parts[0] * 3600 + parts[1] * 60 + parts[2] : 0;
   716→   716→        const left = Math.max(0, Math.round(job.durationSec - done));
   717→   717→        job.eta = left >= 1 ? `${left}s` : "0s";
   718→   718→      }
   719→   719→    }
   720→   720→  });
   721→   721→
   722→   722→  proc.stderr?.on("data", (chunk: Buffer) => {
   723→   723→    stderr += chunk.toString();
   724→   724→    if (stderr.length > 32_000) stderr = stderr.slice(-16_000);
   725→   725→  });
   726→   726→
   727→   727→  proc.on("error", () => {
   728→   728→    if (job.state === "cancelled") return;
   729→   729→    job.state = "error";
   730→   730→    job.error = "INTERNAL";
   731→   731→    pump();
   732→   732→  });
   733→   733→
   734→   734→  proc.on("close", (code) => {
   735→   735→    if (job.state === "cancelled") return;
   736→   736→    if (code === 0) {
   737→   737→      const found = path.join(jobsDir(), `${job.id}.${job.ext}`);
   738→   738→      if (existsSync(found)) {
   739→   739→        job.filePath = found;
   740→   740→        job.sizeBytes = statSync(found).size;
   741→   741→        job.state = "ready";
   742→   742→        job.phase = "done";
   743→   743→        job.progress = 1;
   744→   744→        job.speed = null;
   745→   745→        job.eta = null;
   746→   746→        registerFinishedFile(job);
   747→   747→      } else {
   748→   748→        job.state = "error";
   749→   749→        job.error = "INTERNAL";
   750→   750→      }
   751→   751→    } else {
   752→   752→      const errCode = /403|401/.test(stderr) ? "EXPIRED" : "FETCH_FAILED";
   753→   753→      if (scheduleJobRetry(job, errCode)) {
   754→   754→        pump();
   755→   755→        return;
   756→   756→      }
   757→   757→      job.state = "error";
   758→   758→      job.error = errCode;
   759→   759→    }
   760→   760→    pump();
   761→   761→  });
   762→   762→}
